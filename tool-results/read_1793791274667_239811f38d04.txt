   430→/** Panel de cambios: elige titular que sale + suplente que entra (máx. 3 por equipo). */
   431→function SubsPanel({
   432→  state,
   433→  onConfirm,
   434→  onClose,
   435→}: {
   436→  state: SubState | null
   437→  onConfirm: (outIdx: number, benchIdx: number) => SubResult
   438→  onClose: () => void
   439→}) {
   440→  // el panel solo se muestra en pausa (simulación congelada); el estado lo trae el padre
   441→  const [outIdx, setOutIdx] = useState<number | null>(null)
   442→  const [benchIdx, setBenchIdx] = useState<number | null>(null)
   443→  const [msg, setMsg] = useState('')
   444→
   445→  if (!state) {
   446→    return (
   447→      <div className="fp3d-fade-up rounded-2xl border border-white/12 bg-zinc-950/90 p-6 text-white shadow-2xl">
   448→        Cargando plantilla…
   449→      </div>
   450→    )
   451→  }
   452→
   453→  const outPlayer = state.onField.find((p) => p.idx === outIdx) ?? null
   454→  const benchPlayer = state.bench.find((b) => b.idx === benchIdx) ?? null
   455→  const remaining = state.max - state.used
   456→  const compatible = !outPlayer || !benchPlayer || outPlayer.gk === (benchPlayer.role === 'GK')
   457→  const canConfirm = outPlayer && benchPlayer && compatible && remaining > 0
   458→
   459→  const confirm = () => {
   460→    if (outIdx === null || benchIdx === null) return
   461→    const r = onConfirm(outIdx, benchIdx)
   462→    if (r === 'ok') {
   463→      setMsg('Cambio realizado ✓')
   464→      setOutIdx(null)
   465→      setBenchIdx(null)
   466→    } else if (r === 'limite') {
   467→      setMsg('No quedan cambios disponibles')
   468→    } else {
   469→      setMsg('Cambio inválido (revisá las posiciones)')
   470→    }
   471→  }
   472→
   473→  return (
   474→    <div className="fp3d-fade-up w-[min(94vw,600px)] rounded-2xl border border-white/12 bg-zinc-950/95 p-4 text-white shadow-2xl sm:p-5">
   475→      <div className="flex items-center justify-between">
   476→        <h2 className="text-lg font-black italic uppercase tracking-tight text-amber-300">Cambios</h2>
   477→        <span className="rounded-full bg-white/10 px-2.5 py-1 text-[11px] font-bold">
   478→          Restan <b className={remaining > 0 ? 'text-emerald-400' : 'text-rose-400'}>{remaining}</b> de {state.max}
   479→        </span>
   480→      </div>
   481→      <p className="mt-1 text-[11px] text-white/45">
   482→        Elegí quién sale y quién entra. El suplente entra fresco (resistencia 100).
   483→      </p>
   484→
   485→      <div className="mt-3 grid gap-3 sm:grid-cols-2">
   486→        {/* titulares en cancha */}
   487→        <div>
   488→          <div className="mb-1.5 text-[10px] font-bold uppercase tracking-widest text-white/45">En cancha (sale)</div>

   597→/** Barra de potencia que se actualiza cada frame vía rAF. */
   598→function PowerBar({ engineRef }: { engineRef: React.RefObject<GameEngine | null> }) {
   599→  const ref = useRef<HTMLDivElement>(null)
   600→  const wrapRef = useRef<HTMLDivElement>(null)
   601→  useEffect(() => {
   602→    let raf = 0
   603→    const tick = () => {
   604→      raf = requestAnimationFrame(tick)
   605→      const e = engineRef.current
   606→      const wrap = wrapRef.current
   607→      const bar = ref.current
   608→      if (!e || !wrap || !bar) return
   609→      if (e.isCharging && e.charge > 0.02) {
   610→        wrap.style.opacity = '1'
   611→        bar.style.width = `${Math.round(e.charge * 100)}%`
   612→        bar.style.background = e.charge > 0.8 ? '#f43f5e' : e.charge > 0.5 ? '#fb923c' : '#34d399'
   613→      } else {
   614→        wrap.style.opacity = '0'
   615→      }
   616→    }
   617→    raf = requestAnimationFrame(tick)
   618→    return () => cancelAnimationFrame(raf)
   619→  }, [engineRef])
   620→  return (
   621→    <div
   622→      ref={wrapRef}
   623→      className="pointer-events-none absolute bottom-6 left-1/2 z-20 hidden -translate-x-1/2 rounded-full border border-white/20 bg-black/55 p-1 opacity-0 transition-opacity sm:block"
   624→    >
   625→      <div className="h-2.5 w-52 overflow-hidden rounded-full bg-white/10">
   626→        <div ref={ref} className="h-full w-0 rounded-full bg-emerald-400 transition-[width] duration-75" />
   627→      </div>
   628→    </div>
   629→  )
   630→}
