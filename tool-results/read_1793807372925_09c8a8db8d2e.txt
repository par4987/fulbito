   664→   664→function PowerBar({ engineRef }: { engineRef: React.RefObject<GameEngine | null> }) {
   665→   665→  const ref = useRef<HTMLDivElement>(null)
   666→   666→  const wrapRef = useRef<HTMLDivElement>(null)
   667→   667→  useEffect(() => {
   668→   668→    let raf = 0
   669→   669→    const tick = () => {
   670→   670→      raf = requestAnimationFrame(tick)
   671→   671→      const e = engineRef.current
   672→   672→      const wrap = wrapRef.current
   673→   673→      const bar = ref.current
   674→   674→      if (!e || !wrap || !bar) return
   675→   675→      if (e.isCharging && e.charge > 0.02) {
   676→   676→        wrap.style.opacity = '1'
   677→   677→        bar.style.width = `${Math.round(e.charge * 100)}%`
   678→   678→        bar.style.background = e.charge > 0.8 ? '#f43f5e' : e.charge > 0.5 ? '#fb923c' : '#34d399'
   679→   679→      } else {
   680→   680→        wrap.style.opacity = '0'
   681→   681→      }
   682→   682→    }
   683→   683→    raf = requestAnimationFrame(tick)
   684→   684→    return () => cancelAnimationFrame(raf)
   685→   685→  }, [engineRef])
   686→   686→  return (
   687→   687→    <div
   688→   688→      ref={wrapRef}
   689→   689→      className="pointer-events-none absolute bottom-6 left-1/2 z-20 hidden -translate-x-1/2 rounded-full border border-white/20 bg-black/55 p-1 opacity-0 transition-opacity sm:block"
