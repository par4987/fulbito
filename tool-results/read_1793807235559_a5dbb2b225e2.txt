   179→  setupKickoff(kickingTeam: TeamIndex) {
   180→    this.phase = 'kickoff'
   181→    this.phaseTimer = MATCH_RULES.kickoffFreeze
   182→    this.kickoffTeam = kickingTeam
   183→    this.owner = null
   184→    this.setPieceInfo = null
   185→    this.setPieceTaker = null
   186→    this.setPieceReady = false
   187→    this.ball.reset()
   188→    for (const sq of this.squads) this.placeFormation(sq, kickingTeam)
   189→    const sqK = this.squads[kickingTeam]
   190→    let taker = sqK.players[9] // FW central
   191→    if (!taker || taker.sentOff) {
   192→      taker = sqK.players.find((p) => !p.isGK && !p.sentOff) ?? taker
   193→    }
   194→    if (taker) {
   195→      taker.teleport(PITCH.L / 2 - this.squads[kickingTeam].attackDir * 0.8, PITCH.W / 2)
   196→      this.gainControl(taker)
   197→    }
   198→    this.controlled = kickingTeam === 0 ? (taker ?? null) : null
   199→    this.audio.whistle(1)
   200→  }

   202→  private placeFormation(sq: Squad, kickingTeam: TeamIndex) {
   203→    for (const p of sq.players) {
   204→      if (p.sentOff) continue // el expulsado permanece fuera del campo
   205→      let wx = sq.attackDir > 0 ? p.ax * PITCH.L : (1 - p.ax) * PITCH.L
   206→      let wz = p.az * PITCH.W
   207→      // todos en su propia mitad en el saque
   208→      if (sq.attackDir > 0) wx = Math.min(wx, PITCH.L / 2 - 0.5)
   209→      else wx = Math.max(wx, PITCH.L / 2 + 0.5)
   210→      // rivales fuera del círculo central
   211→      if (sq.attackDir !== this.squads[kickingTeam].attackDir) {
   212→        const d = dist2d(wx, wz, PITCH.L / 2, PITCH.W / 2)
   213→        if (d < PITCH.circleR + 1) {
   214→          const ux = (wx - PITCH.L / 2) / (d || 1)
   215→          const uz = (wz - PITCH.W / 2) / (d || 1)
   216→          wx = PITCH.L / 2 + ux * (PITCH.circleR + 1.5)
   217→          wz = PITCH.W / 2 + uz * (PITCH.circleR + 1.5)
   218→        }
   219→      }
   220→      p.teleport(clamp(wx, 1, PITCH.L - 1), clamp(wz, 1, PITCH.W - 1))
   221→      p.targetX = p.x
   222→      p.targetZ = p.z
   223→    }
   224→  }

   226→  private setRestart(info: RestartInfo) {
   227→    this.phase = 'restart'
   228→    this.phaseTimer = MATCH_RULES.restartFreeze
   229→    this.restart = info
   230→    this.ball.reset(info.x, info.z)
   231→    this.owner = null
   232→    const squad = this.squads[info.team]
   233→    let best: PlayerEntity | null = null
   234→    let bestD = Infinity
   235→    for (const p of squad.players) {
   236→      if (p.sentOff) continue
   237→      if (info.type === 'goalkick' ? !p.isGK : p.isGK) continue
   238→      const d = dist2d(p.x, p.z, info.x, info.z)
   239→      if (d < bestD) {
   240→        bestD = d
   241→        best = p
   242→      }
   243→    }
   244→    if (best) {
   245→      best.teleport(info.x - Math.sign(info.x - PITCH.L / 2) * 0.9, info.z)
   246→      this.gainControl(best)
   247→      if (info.team === 0) this.controlled = best
   248→      else this.controlled = null
   249→    }
   250→  }

   252→  private handleGoal(scoringTeam: TeamIndex) {
   253→    const sq = this.squads[scoringTeam]
   254→    sq.score++
   255→    if (this.pendingShot && this.shotTeam === scoringTeam) {
   256→      sq.onTarget++
   257→    }
   258→    const scorer = this.lastTouchPlayer
   259→    const own = scorer && scorer.team !== scoringTeam
   260→    const name = scorer ? scorer.data.name : 'desconocido'
   261→    this.celebrateScorer = own ? `${name} (autogol)` : name
   262→    this.celebrateTeam = scoringTeam
   263→    this.phase = 'goal'
   264→    this.phaseTimer = MATCH_RULES.goalCelebration
   265→    this.owner = null
   266→    this.pendingShot = false
   267→    this.audio.goalRoar()
   268→    this.excitement = 1
   269→    this.addEvent(`¡GOL de ${this.celebrateScorer} (${sq.roster.def.short})!`, 'goal')
   270→  }

   329→  private kickBall(p: PlayerEntity, dx: number, dz: number, speed: number, vz: number) {
   330→    this.ball.kick(dx, dz, speed, vz)

   340→  /** Pase raso o alto hacia el mejor compañero según dirección deseada. */
   341→  doPass(p: PlayerEntity, aimX: number, aimZ: number, lofted = false) {
   342→    if (p.kickCooldown > 0) return
   343→    const receiver = this.pickPassTarget(p, aimX, aimZ, lofted)
   344→    if (!receiver) {
   345→      // despeje hacia adelante
   346→      const dir = this.squads[p.team].attackDir
   347→      this.kickBall(p, dir, (Math.random() - 0.5) * 20, 16, 4)
   348→      return
   349→    }
   350→    const err = (1 - p.data.skill) * this.diff.passErr * (lofted ? 1.6 : 1)
   351→    const noise = (Math.random() - 0.5) * 2 * err
   352→    const cs = Math.cos(noise)
   353→    const sn = Math.sin(noise)
   354→    const dist = dist2d(p.x, p.z, receiver.x, receiver.z)
   355→    const speed = lofted ? clamp(10 + dist * 0.42, 13, 20) : clamp(9 + dist * 0.55, 10, 21)
   356→    // punto adelantado hacia donde corre el receptor
   357→    const lead = lofted ? 0.35 : 0.45
   358→    let dx = receiver.x + receiver.vx * lead - p.x
   359→    let dz = receiver.z + receiver.vz * lead - p.z
   360→    const rx = dx * cs - dz * sn
   361→    const rz = dx * sn + dz * cs
   362→    const vz = lofted ? clamp(dist * 0.32, 3, 8.5) : 0.35
   363→    this.kickBall(p, rx, rz, speed, vz)
   364→    if (p.team === 0 && !lofted) this.controlled = receiver
   365→    if (p.team === 1) this.aiPassHint = receiver
   366→  }

   368→  private pickPassTarget(p: PlayerEntity, aimX: number, aimZ: number, lofted: boolean): PlayerEntity | null {
   369→    const aimLen = Math.hypot(aimX, aimZ)
   370→    const nx = aimLen > 0.1 ? aimX / aimLen : Math.sin(p.dir)
   371→    const nz = aimLen > 0.1 ? aimZ / aimLen : Math.cos(p.dir)
   372→    const squad = this.squads[p.team]
   373→    let best: PlayerEntity | null = null
   374→    let bestScore = -Infinity
   375→    for (const mate of squad.players) {
   376→      if (mate === p || mate.isGK || mate.sentOff) continue
   377→      const dx = mate.x - p.x
   378→      const dz = mate.z - p.z
   379→      const d = Math.hypot(dx, dz)
   380→      if (d < 2.5 || d > (lofted ? 55 : 34)) continue
   381→      const dot = (dx / d) * nx + (dz / d) * nz
   382→      if (!lofted && dot < 0.2) continue
   383→      const open = this.closestOppDist(mate)
   384→      let lanePen = 0
   385→      if (!lofted) lanePen = this.laneBlocked(p, mate) ? 2.5 : 0
   386→      const score = dot * 3 + open * 0.35 - d * 0.045 - lanePen
   387→      if (score > bestScore) {
   388→        bestScore = score
   389→        best = mate
   390→      }
   391→    }
   392→    return best
   393→  }

   395→  private closestOppDist(p: PlayerEntity): number {
   396→    const opp = this.squads[p.team === 0 ? 1 : 0].players
   397→    let d = Infinity
   398→    for (const o of opp) {
   399→      if (o.sentOff) continue
   400→      d = Math.min(d, dist2d(o.x, o.z, p.x, p.z))
   401→    }
   402→    return d
   403→  }
   404→
   405→  private laneBlocked(from: PlayerEntity, to: PlayerEntity): boolean {
   406→    const opp = this.squads[from.team === 0 ? 1 : 0].players
   407→    const dx = to.x - from.x
   408→    const dz = to.z - from.z
   409→    const len = Math.hypot(dx, dz) || 1
   410→    for (const o of opp) {
   411→      if (o.sentOff) continue
   412→      const t = clamp(((o.x - from.x) * dx + (o.z - from.z) * dz) / (len * len), 0.1, 0.9)
   413→      const px = from.x + dx * t
   414→      const pz = from.z + dz * t
   415→      if (dist2d(o.x, o.z, px, pz) < 1.3) return true
   416→    }
   417→    return false
   418→  }

   420→  /** Disparo al arco. power 0..1, aim = componente perpendicular del input. */
   421→  doShoot(p: PlayerEntity, power: number, aim: number) {
   422→    if (p.kickCooldown > 0) return
   423→    const gx = this.goalOf(p.team)
   424→    const dirX = gx > p.x ? 1 : -1
   425→    const dist = Math.abs(gx - p.x)
   426→    const noise = (1 - p.data.skill) * this.diff.shootErr * (2 + dist * 0.06) + power * 0.04
   427→    const angle = (Math.random() - 0.5) * 2 * noise
   428→    let dz = GOAL.zCenter - p.z + aim * 3.4
   429→    let dx = dirX * Math.max(8, dist)
   430→    const cs = Math.cos(angle)
   431→    const sn = Math.sin(angle)
   432→    const rx = dx * cs - dz * sn
   433→    const rz = dx * sn + dz * cs
   434→    const speed = 18 + power * 12
   435→    const vz = 0.5 + power * 3.1
   436→    this.kickBall(p, rx, rz, speed, vz)
   437→    this.squads[p.team].shots++
   438→    this.pendingShot = true
   439→    this.shotTeam = p.team
   440→    this.shotWindow = 1.3
   441→  }

   443→  /** Entrada (presión): intenta robar el balón al dueño cercano. Puede cometer falta. */
   444→  tryTackle(p: PlayerEntity) {
   445→    if (p.tackleCd > 0 || p.stunTimer > 0) return
   446→    p.tackleCd = PLAYER.tackleCooldown
   447→    const owner = this.owner
   448→    if (owner && owner.isGK && this.gkHold > 0) return // balón en manos del portero
   449→    if (owner && owner.team !== p.team && dist2d(p.x, p.z, owner.x, owner.z) < PLAYER.tackleRange) {
   450→      const prob = 0.45 + p.data.skill * 0.25 + this.diff.tackle * 0.15 - owner.data.skill * 0.2
   451→      if (Math.random() < prob) {
   452→        this.loseControl()
   453→        this.ball.kick(p.vx + Math.sin(p.dir) * 4, p.vz + Math.cos(p.dir) * 4, 7, 0.4)
   454→        this.ball.lastTouch = p.team
   455→        this.lastTouchPlayer = p
   456→        owner.stunTimer = 0.3
   457→        this.audio.kick(0.3)
   458→      } else {
   459→        p.stunTimer = 0.35
   460→        // entrada fallada con contacto: puede ser sancionada
   461→        if (Math.random() < FOUL.standingFoulChance) {
   462→          this.awardFoul(p, owner)
   463→        }
   464→      }
   465→    } else if (!owner && dist2d(p.x, p.z, this.ball.x, this.ball.z) < PLAYER.tackleRange) {
   466→      // empujón al balón libre
   467→      this.ball.kick(Math.sin(p.dir), Math.cos(p.dir), 8, 0.5)
   468→      this.ball.lastTouch = p.team
   469→      this.lastTouchPlayer = p
   470→      p.kickCooldown = 0.3
   471→    }
   472→  }

   505→  /**
   506→   * Realiza un cambio: el titular conserva su puesto en la formación y el rig 3D;
   507→   * solo se rebrandea (nombre, dorsal, atributos) y recupera resistencia.
   508→   * Portero solo por portero (el rig del ARQ es distinto y su IA es especial).
   509→   */
   510→  makeSubstitution(team: TeamIndex, outIdx: number, benchIdx: number): SubResult {
   511→    if (this.phase === 'full') return 'invalido'
   512→    const sq = this.squads[team]
   513→    if (sq.subsUsed >= MATCH_RULES.maxSubs) return 'limite'
   514→    const out = sq.players[outIdx]
   515→    const sub = sq.bench[benchIdx]
   516→    if (!out || !sub || sq.benchUsed[benchIdx]) return 'invalido'
   517→    if (out.sentOff) return 'invalido' // al expulsado no lo reemplaza nadie
   518→    if (out.isGK !== (sub.role === 'GK')) return 'invalido'
   519→    const oldName = out.data.name
   520→    out.data = sub
   521→    out.stamina = 100
   522→    out.yellowCards = 0 // el que entra llega sin tarjetas
   523→    sq.benchUsed[benchIdx] = true
   524→    sq.subsUsed++
   525→    this.addEvent(`Cambio ${sq.roster.def.short}: entra ${sub.name} por ${oldName}`, 'info')
   526→    return 'ok'
   527→  }

   531→  /** La CPU reemplaza a sus jugadores más cansados a partir del minuto 50 (hasta 5 cambios). */
   532→  private tryCpuSub() {
   533→    const sq = this.squads[1]
   534→    if (sq.subsUsed >= MATCH_RULES.maxSubs) return
   535→    if (this.minute() < 50) return
   536→    // puede hacer hasta dos cambios por revisión si hay varios fundidos
   537→    for (let n = 0; n < 2; n++) {
   538→      if (sq.subsUsed >= MATCH_RULES.maxSubs) return
   539→      let worst: PlayerEntity | null = null
   540→      let worstIdx = -1
   541→      for (let i = 0; i < sq.players.length; i++) {
   542→        const p = sq.players[i]
   543→        if (p.isGK || p.sentOff) continue
   544→        if (!worst || p.stamina < worst.stamina) {
   545→          worst = p
   546→          worstIdx = i
   547→        }
   548→      }
   549→      if (!worst || worst.stamina > 62) return
   550→      let bi = sq.bench.findIndex((b, i) => !sq.benchUsed[i] && b.role === worst!.data.role)
   551→      if (bi < 0) bi = sq.bench.findIndex((b, i) => !sq.benchUsed[i] && b.role !== 'GK')
   552→      if (bi < 0) return
   553→      this.makeSubstitution(1, worstIdx, bi)
   554→    }
   555→  }

   559→  update(dt: number) {
   560→    this.excitement = Math.max(0, this.excitement - dt * 0.35)
   561→    this.audio.setExcitement(this.excitement * 0.8)
   562→    this.shotWindow = Math.max(0, this.shotWindow - dt)
   563→    if (this.shotWindow === 0) this.pendingShot = false
   564→
   565→    // temporizador del overlay de tarjetas
   566→    if (this.cardTimer > 0) {
   567→      this.cardTimer -= dt
   568→      if (this.cardTimer <= 0) this.cardOverlay = null
   569→    }
   570→
   571→    // cambios automáticos de la CPU cuando el cansancio aprieta
   572→    // (se revisan en toda fase: las bolas paradas son momento de cambios)
   573→    if (this.phase !== 'full') {
   574→      this.cpuSubTimer -= dt
   575→      if (this.cpuSubTimer <= 0) {
   576→        this.cpuSubTimer = 8
   577→        this.tryCpuSub()
   578→      }
   579→    }
   580→
   581→    switch (this.phase) {
   582→      case 'kickoff':
   583→      case 'restart':
   584→      case 'half': {
   585→        this.phaseTimer -= dt
   586→        if (this.phase === 'half') {
   587→          if (this.phaseTimer <= 0) {
   588→            this.half = 2
   589→            for (const sq of this.squads) sq.attackDir = (sq.attackDir * -1) as 1 | -1
   590→            // descanso: los jugadores recuperan resistencia en el mediempo
   591→            for (const sq of this.squads)
   592→              for (const p of sq.players) p.stamina = Math.min(100, p.stamina + MATCH_RULES.halfTimeRecovery)
   593→            this.setupKickoff(1)
   594→            this.addEvent('Comienza el segundo tiempo', 'info')
   595→          }
   596→          break
   597→        }
   598→        // congelados: solo animación
   599→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   600→        this.ballPrevX = this.ball.x
   601→        if (this.phaseTimer <= 0) {
   602→          this.phase = 'play'
   603→        }
   604→        break
   605→      }
   606→      case 'goal': {
   607→        this.phaseTimer -= dt
   608→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   609→        if (this.phaseTimer <= 0) {
   610→          this.celebrateScorer = null
   611→          this.celebrateTeam = null
   612→          this.setupKickoff(this.lastTouchTeamConceding())
   613→        }
   614→        break
   615→      }
   616→      case 'play':
   617→        this.updatePlay(dt)
   618→        break
   619→      case 'freekick':
   620→      case 'penalty':
   621→        this.updateSetPiece(dt)
   622→        break
   623→      case 'full':
   624→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   625→        break
   626→    }
   627→    this.input.endFrame()
   628→  }

   631→    if (!this.lastTouchPlayer) return this.kickoffTeam === 0 ? 1 : 0
   632→    // el equipo que recibió el gol saca
   633→    return this.lastTouchPlayer.team === 0 ? 1 : 0
   634→  }

   636→  private updatePlay(dt: number) {
   637→    this.clock += dt
   638→    this.ballPrevX = this.ball.x
   639→
   640→    this.updateControlledInput(dt)
   641→    updateMatchAI(this, dt)
   642→    for (const sq of this.squads) {
   643→      for (const p of sq.players) p.update(dt)
   644→    }
   645→    this.separatePlayers()
   646→    // faltas: barridas que golpean rival antes que el balón (detienen la jugada)
   647→    if (this.checkFouls()) return
   648→    this.updateBallPossession(dt)
   649→    this.checkOutOfBounds()
   650→
   651→    if (this.owner) this.squads[this.owner.team].possession += dt
   652→
   653→    if (this.half === 1 && this.clock >= this.durationSec / 2) {
   654→      this.finishHalf()
   655→    } else if (this.half === 2 && this.clock >= this.durationSec && !this.finished) {
   656→      this.goFullTime()
   657→    }
   658→  }

   660→  /** Convierte la entrada humana en acciones del jugador controlado. */
   661→  private updateControlledInput(dt: number) {
   662→    const inp = this.input
   663→    // auto-cambio de jugador
   664→    const team0 = this.squads[0]
   665→    if (!this.owner || this.owner.team !== 0) {
   666→      this.switchHysteresis += dt
   667→      if (this.switchHysteresis > 0.18 || !this.controlled || this.controlled.team !== 0) {
   668→        const fut = this.ball.future(0.25)
   669→        let best: PlayerEntity | null = null
   670→        let bestD = Infinity
   671→        const nearOwnBox = Math.abs(this.ball.x - this.ownGoalOf(0)) < 20
   672→        for (const p of team0.players) {
   673→          if (p.sentOff) continue
   674→          if (p.isGK && !nearOwnBox) continue
   675→          const d = dist2d(p.x, p.z, fut.x, fut.z)
   676→          if (d < bestD) {
   677→            bestD = d
   678→            best = p
   679→          }
   680→        }
   681→        if (best && best !== this.controlled) this.controlled = best
   682→        this.switchHysteresis = 0
   683→      }
   684→    } else {
   685→      this.switchHysteresis = 0
   686→      this.controlled = this.owner
   687→    }
   688→
   689→    const c = this.controlled
   690→    if (!c) return
   691→    c.speedMult = this.diff.userBoost
   692→
   693→    // movimiento relativo a cámara (el engine lo traduce antes: aquí ya es mundo)
   694→    const mx = inp.move.x
   695→    const my = inp.move.y
   696→    const world = this.engineMoveToWorld(mx, my)
   697→    const moving = Math.hypot(world.x, world.z) > 0.1
   698→    c.sprinting = inp.sprint && moving
   699→    if (moving) {
   700→      const len = Math.hypot(world.x, world.z)
   701→      c.desiredVx = (world.x / len) * c.maxSpeed()
   702→      c.desiredVz = (world.z / len) * c.maxSpeed()
   703→    } else {
   704→      c.desiredVx = 0
   705→      c.desiredVz = 0
   706→    }
   707→
   708→    const isOwner = this.owner === c
   709→    if (isOwner) {
   710→      // disparo con carga
   711→      if (inp.isHeld('shoot')) {
   712→        this.charging = true
   713→        this.charge = Math.min(1, this.charge + dt / 0.85)
   714→        if (this.charge >= 1) {
   715→          this.releaseShot(c)
   716→        }
   717→      } else if (this.charging) {
   718→        this.releaseShot(c)
   719→      }
   720→      if (inp.consumePressed('pass')) {
   721→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   722→        this.doPass(c, aim.x, aim.z, false)
   723→        this.charge = 0
   724→        this.charging = false
   725→      }
   726→      if (inp.consumePressed('lob')) {
   727→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   728→        this.doPass(c, aim.x, aim.z, true)
   729→      }
   730→    } else {
   731→      // defensa
   732→      this.charge = 0
   733→      this.charging = false
   734→      if (inp.consumePressed('pass')) this.tryTackle(c)
   735→      if (inp.consumePressed('shoot')) {
   736→        const dx = moving ? world.x : Math.sin(c.dir)
   737→        const dz = moving ? world.z : Math.cos(c.dir)
   738→        this.slideTackle(c, dx, dz)
   739→      }
   740→      if (inp.consumePressed('lob') || inp.consumePressed('switch')) {
   741→        // cambio manual: siguiente más cercano
   742→        const fut = this.ball.future(0.25)
   743→        const others = team0.players
   744→          .filter((p) => !p.isGK && !p.sentOff && p !== c)
   745→          .sort((a, b) => dist2d(a.x, a.z, fut.x, fut.z) - dist2d(b.x, b.z, fut.x, fut.z))
   746→        if (others[0]) this.controlled = others[0]
   747→      }
   748→    }
   749→  }

   751→  private releaseShot(c: PlayerEntity) {
   752→    const world = this.engineMoveToWorld(this.input.move.x, this.input.move.y)
   753→    // input vertical en mundo: arriba = palo lejano, abajo = palo cercano
   754→    const aimProj = clamp(world.z, -1, 1)
   755→    this.doShoot(c, Math.max(0.25, this.charge), aimProj)
   756→    this.charge = 0
   757→    this.charging = false
   758→  }

   760→  /** Traduce input de pantalla a mundo según el modo de cámara (seteado por el engine). */
   761→  cameraMode: CameraMode = 'broadcast'
   762→  attackDirUser: 1 | -1 = 1
   763→  private engineMoveToWorld(x: number, y: number): { x: number; z: number } {
   764→    if (this.cameraMode === 'arco') {
   765→      // cámara tras el arco que ataca el usuario
   766→      const f = this.attackDirUser
   767→      return { x: y * f, z: -x * f }
   768→    }
   769→    return { x, z: -y }
   770→  }

   774→  /**
   775→   * Detecta faltas de barrida: si el jugador deslizándose golpea a un rival
   776→   * sin haber tocado antes el balón, el árbitro pita la falta. Devuelve true
   777→   * si pitó algo (la jugada se detuvo este frame).
   778→   */
   779→  private checkFouls(): boolean {
   780→    if (this.phase !== 'play') return false
   781→    const ball = this.ball
   782→    const all = [...this.squads[0].players, ...this.squads[1].players]
   783→    for (const p of all) {
   784→      if (p.sentOff || p.slideTimer <= 0.06) continue
   785→      const ballDist = dist2d(p.x, p.z, ball.x, ball.z)
   786→      if (ballDist < PLAYER.tackleRange) {
   787→        // barrida limpia: llega primero al balón y lo despeja (una sola vez)
   788→        if (p.kickCooldown <= 0) {
   789→          const dx = ball.x - p.x
   790→          const dz = ball.z - p.z
   791→          const len = Math.hypot(dx, dz) || 1
   792→          this.loseControl()
   793→          this.ball.kick(dx / len + p.vx * 0.08, dz / len + p.vz * 0.08, 9.5, 1.4)
   794→          this.ball.lastTouch = p.team
   795→          this.lastTouchPlayer = p
   796→          p.kickCooldown = 0.5
   797→          const owner = this.owner
   798→          if (owner && owner.team !== p.team && dist2d(p.x, p.z, owner.x, owner.z) < FOUL.contactRadius) {
   799→            owner.stunTimer = 0.55 // la barrida limpia también derriba
   800→          }
   801→          this.audio.kick(0.45)
   802→        }
   803→        continue
   804→      }
   805→      for (const o of all) {
   806→        if (o.sentOff || o.team === p.team) continue
   807→        if (dist2d(p.x, p.z, o.x, o.z) < FOUL.contactRadius) {
   808→          this.awardFoul(p, o)
   809→          return true
   810→        }
   811→      }
   812→    }
   813→    return false
   814→  }

   816→  /** Sanciona una falta: tarjeta si corresponde y tiro libre o penal. */
   817→  private awardFoul(offender: PlayerEntity, victim: PlayerEntity) {
   818→    const sq = this.squads[offender.team]
   819→    sq.fouls++
   820→    const speed = offender.speedNow
   821→    // ¿llegó desde atrás? (proyección del ofensor sobre la dirección del pie del victimario)
   822→    const relX = offender.x - victim.x
   823→    const relZ = offender.z - victim.z
   824→    const relLen = Math.hypot(relX, relZ) || 1
   825→    const behind = (Math.sin(victim.dir) * relX + Math.cos(victim.dir) * relZ) / relLen < -0.35
   826→
   827→    victim.stunTimer = FOUL.victimStun
   828→    this.loseControl()
   829→    const fx = clamp(victim.x, 0.5, PITCH.L - 0.5)
   830→    const fz = clamp(victim.z, 0.5, PITCH.W - 0.5)
   831→    this.ball.reset(fx, fz)
   832→
   833→    // severidad → tarjeta (velocidad de la entrada + jugada por detrás)
   834→    const sev = speed / 17 + (behind ? 0.4 : 0) + Math.random() * 0.12
   835→    let card: 'none' | 'yellow' | 'red' = 'none'
   836→    if (sev > 0.85) card = 'yellow'
   837→    if (sev > 1.22) card = 'red'
   838→    if (card === 'yellow' && offender.yellowCards >= 1) card = 'red' // segunda amarilla
   839→    if (offender.isGK && card === 'red') card = 'yellow' // evitamos dejar el arco vacío
   840→
   841→    this.addEvent(`Falta de ${offender.data.name} (${sq.roster.def.short})`, 'info')
   842→    if (card !== 'none') this.applyCard(offender, card)
   843→    this.audio.whistle(1)
   844→    this.excitement = Math.max(this.excitement, 0.45)
   845→
   846→    // ¿dentro del área propia del que cometió la falta? → penal
   847→    const defGoalX = this.ownGoalOf(offender.team)
   848→    const inBox = Math.abs(fx - defGoalX) < PITCH.boxD && Math.abs(fz - GOAL.zCenter) < PITCH.boxW / 2
   849→    if (inBox) this.setupPenalty(victim.team)
   850→    else this.setupFreeKick(victim.team, fx, fz)
   851→  }

   853→  /** Muestra tarjeta, la registra y expulsa si es roja (o segunda amarilla). */
   854→  private applyCard(p: PlayerEntity, card: 'yellow' | 'red') {
   855→    const sq = this.squads[p.team]
   856→    if (card === 'yellow') {
   857→      p.yellowCards++
   858→      sq.yellows++
   859→      this.cardOverlay = { kind: 'amarilla', player: p.data.name, team: p.team }
   860→      this.cardTimer = 2.6
   861→      this.addEvent(`Amarilla para ${p.data.name} (${sq.roster.def.short})`, 'info')
   862→      if (p.yellowCards >= 2) {
   863→        this.cardOverlay = { kind: 'roja', player: p.data.name, team: p.team }
   864→        this.addEvent(`¡Segunda amarilla! ${p.data.name} expulsado`, 'info')
   865→        this.expel(p)
   866→      }
   867→    } else {
   868→      this.cardOverlay = { kind: 'roja', player: p.data.name, team: p.team }
   869→      this.cardTimer = 3.0
   870→      this.addEvent(`¡Roja directa! ${p.data.name} (${sq.roster.def.short}) expulsado`, 'info')
   871→      this.expel(p)
   872→    }
   873→  }

   875→  private expel(p: PlayerEntity) {
   876→    this.squads[p.team].reds++
   877→    p.sendOff()
   878→    if (this.owner === p) this.loseControl()
   879→    if (this.controlled === p) {
   880→      let best: PlayerEntity | null = null
   881→      let bestD = Infinity
   882→      for (const m of this.squads[p.team].players) {
   883→        if (m === p || m.sentOff || m.isGK) continue
   884→        const d = dist2d(m.x, m.z, this.ball.x, this.ball.z)
   885→        if (d < bestD) {
   886→          bestD = d
   887→          best = m
   888→        }
   889→      }
   890→      if (best) this.controlled = best
   891→    }
   892→    this.audio.whistle(2)
   893→  }

   895→  /** Arma el tiro libre: ejecutante detrás del balón y barrera rival a 9,15 m. */
   896→  private setupFreeKick(team: TeamIndex, x: number, z: number) {
   897→    this.phase = 'freekick'
   898→    this.phaseTimer = FOUL.freeKickFreeze
   899→    this.setPieceInfo = { type: 'freekick', team, x, z }
   900→    this.setPieceReady = false
   901→    this.ball.reset(x, z)
   902→    this.owner = null
   903→    this.pendingShot = false
   904→
   905→    const sq = this.squads[team]
   906→    let taker: PlayerEntity | null = null
   907→    let bestScore = -Infinity
   908→    for (const p of sq.players) {
   909→      if (p.isGK || p.sentOff) continue
   910→      const score = p.data.skill * 6 - dist2d(p.x, p.z, x, z) * 0.5
   911→      if (score > bestScore) {
   912→        bestScore = score
   913→        taker = p
   914→      }
   915→    }
   916→    this.setPieceTaker = taker
   917→    if (taker) {
   918→      const gx = this.goalOf(team)
   919→      const dx = gx - x
   920→      const dz = GOAL.zCenter - z
   921→      const d = Math.hypot(dx, dz) || 1
   922→      taker.teleport(x - (dx / d) * 1.1, z - (dz / d) * 1.1)
   923→      taker.dir = Math.atan2(dx, dz)
   924→      this.gainControl(taker)
   925→      this.gkHold = 0
   926→    }
   927→    this.buildWall(team === 0 ? 1 : 0, x, z)
   928→    this.controlled = team === 0 ? taker : null
   929→  }

   931→  private buildWall(defTeam: TeamIndex, x: number, z: number) {
   932→    const gx = this.ownGoalOf(defTeam)
   933→    const dx = gx - x
   934→    const dz = GOAL.zCenter - z
   935→    const d = Math.hypot(dx, dz) || 1
   936→    const ux = dx / d
   937→    const uz = dz / d
   938→    const wx = x + ux * FOUL.wallDist
   939→    const wz = z + uz * FOUL.wallDist
   940→    const px = -uz
   941→    const pz = ux
   942→    const defs = this.squads[defTeam].players
   943→      .filter((p) => !p.isGK && !p.sentOff)
   944→      .sort((a, b) => dist2d(a.x, a.z, wx, wz) - dist2d(b.x, b.z, wx, wz))
   945→      .slice(0, FOUL.wallPlayers)
   946→    defs.forEach((p, i) => {
   947→      const off = (i - (defs.length - 1) / 2) * 0.72
   948→      p.teleport(clamp(wx + px * off, 0.6, PITCH.L - 0.6), clamp(wz + pz * off, 0.6, PITCH.W - 0.6))
   949→      p.dir = Math.atan2(x - p.x, z - p.z)
   950→      p.targetX = p.x
   951→      p.targetZ = p.z
   952→    })
   953→  }

   955→  /** Penal: ejecutante, portero en la línea y todos los demás fuera del área. */
   956→  private setupPenalty(team: TeamIndex) {
   957→    this.phase = 'penalty'
   958→    this.phaseTimer = FOUL.penaltyFreeze
   959→    const defGoalX = this.ownGoalOf(team === 0 ? 1 : 0)
   960→    const dir = defGoalX < PITCH.L / 2 ? 1 : -1
   961→    const px = defGoalX + dir * PITCH.spot
   962→    this.setPieceInfo = { type: 'penalty', team, x: px, z: GOAL.zCenter }
   963→    this.setPieceReady = false
   964→    this.addEvent(`¡PENAL para ${this.squads[team].roster.def.short}!`, 'chance')
   965→    this.ball.reset(px, GOAL.zCenter)
   966→    this.owner = null
   967→    this.pendingShot = false
   968→
   969→    const sq = this.squads[team]
   970→    let taker: PlayerEntity | null = null
   971→    let bestSkill = -1
   972→    for (const p of sq.players) {
   973→      if (p.isGK || p.sentOff) continue
   974→      if (p.data.skill > bestSkill) {
   975→        bestSkill = p.data.skill
   976→        taker = p
   977→      }
   978→    }
   979→    this.setPieceTaker = taker
   980→    if (taker) {
   981→      taker.teleport(px - dir * 2.2, GOAL.zCenter + 0.4)
   982→      taker.dir = Math.atan2(defGoalX - taker.x, GOAL.zCenter - taker.z)
   983→    }
   984→    const gk = this.squads[team === 0 ? 1 : 0].players[0]

   988→    let side = 1

   990→    for (const other of [...this.squads[0].players, ...this.squads[1].players]) {

  1007→  /** Bucle de las fases de tiro libre / penal: congelación → ejecución → juego. */
  1008→  private updateSetPiece(dt: number) {
  1009→    const sp = this.setPieceInfo
  1010→    if (!sp) {
  1011→      this.resumeFromSetPiece()
  1012→      return
  1013→    }
  1014→    for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
  1015→    this.ballPrevX = this.ball.x
  1016→    this.phaseTimer -= dt
  1017→    if (this.phaseTimer > 0) return
  1018→
  1019→    if (!this.setPieceReady) {
  1020→      this.setPieceReady = true
  1021→      this.phaseTimer = FOUL.setPieceTimeout
  1022→      return
  1023→    }
  1024→
  1025→    const taker = this.setPieceTaker
  1026→    if (!taker) {
  1027→      this.resumeFromSetPiece()
  1028→      return
  1029→    }
  1030→    if (sp.team === 1) {
  1031→      // la CPU ejecuta tras un breve tiempo de preparación
  1032→      if (this.phaseTimer <= FOUL.setPieceTimeout - FOUL.cpuSetPieceDelay) this.executeSetPiece('auto')
  1033→    } else {
  1034→      this.updateSetPieceHumanInput(dt)
  1035→      if (this.setPieceInfo && this.phaseTimer <= 0) this.executeSetPiece('auto')
  1036→    }
  1037→  }

  1039→  /** Entrada humana durante tiro libre / penal propio (jugador fijo, apuntar y ejecutar). */
  1040→  private updateSetPieceHumanInput(dt: number) {
  1041→    const sp = this.setPieceInfo
  1042→    const taker = this.setPieceTaker
  1043→    if (!sp || !taker) return
  1044→    const inp = this.input
  1045→    if (inp.isHeld('shoot')) {
  1046→      this.charging = true
  1047→      this.charge = Math.min(1, this.charge + dt / 0.85)
  1048→    } else if (this.charging) {
  1049→      const world = this.engineMoveToWorld(inp.move.x, inp.move.y)
  1050→      const aim = clamp(world.z, -1, 1)
  1051→      this.doShoot(taker, Math.max(0.3, this.charge), aim)
  1052→      if (sp.type === 'penalty') this.gkGuessDive(taker)
  1053→      this.charge = 0
  1054→      this.charging = false
  1055→      this.resumeFromSetPiece()
  1056→      return
  1057→    }
  1058→    if (inp.consumePressed('pass') || inp.consumePressed('lob')) {
  1059→      const lofted = inp.consumePressed('lob')
  1060→      const world = this.engineMoveToWorld(inp.move.x, inp.move.y)
  1061→      const moving = Math.hypot(world.x, world.z) > 0.1
  1062→      const aim = moving ? world : { x: this.squads[0].attackDir, z: 0 }
  1063→      this.doPass(taker, aim.x, aim.z, lofted)
  1064→      this.charge = 0
  1065→      this.charging = false
  1066→      this.resumeFromSetPiece()
  1067→    }
  1068→  }

  1071→  private executeSetPiece(mode: 'auto') {
  1072→    const sp = this.setPieceInfo

  1075→      this.resumeFromSetPiece()

  1079→    const distG = dist2d(taker.x, taker.z, gx, GOAL.zCenter)

  1091→    }

  1093→    this.charging = false
  1094→    this.resumeFromSetPiece()

  1114→  private separatePlayers() {
  1115→    const all = [...this.squads[0].players, ...this.squads[1].players].filter((p) => !p.sentOff)
  1116→    for (let i = 0; i < all.length; i++) {
  1117→      for (let j = i + 1; j < all.length; j++) {
  1118→        const a = all[i]
  1119→        const b = all[j]
  1120→        const dx = b.x - a.x
  1121→        const dz = b.z - a.z
  1122→        const d = Math.hypot(dx, dz)
  1123→        const min = PLAYER.radius * 2
  1124→        if (d > 0.001 && d < min) {
  1125→          const push = (min - d) / 2
  1126→          const ux = dx / d
  1127→          const uz = dz / d
  1128→          a.x -= ux * push
  1129→          a.z -= uz * push
  1130→          b.x += ux * push
  1131→          b.z += uz * push
  1132→        }
  1133→      }
  1134→    }

  1137→  private updateBallPossession(dt: number) {
  1138→    const ball = this.ball
  1139→    const postHit = ball.update(dt)
  1140→    if (postHit === 'post') {
  1141→      this.audio.post()
  1142→      this.addEvent('¡Al palo!', 'chance')
  1143→      this.excitement = Math.max(this.excitement, 0.6)
  1144→    } else if (postHit === 'crossbar') {
  1145→      this.audio.post()
  1146→      this.addEvent('¡Al travesaño!', 'chance')
  1147→      this.excitement = Math.max(this.excitement, 0.6)
  1148→    }
  1149→
  1150→    const owner = this.owner
  1151→    if (owner) {
  1152→      if (owner.isGK && this.gkHold > 0) {
  1153→        this.gkHold -= dt
  1154→        // balón en las manos
  1155→        ball.x = owner.x + Math.sin(owner.dir) * 0.35
  1156→        ball.z = owner.z + Math.cos(owner.dir) * 0.35
  1157→        ball.y = 1.05
  1158→        ball.stop()
  1159→        const gkTeam = owner.team
  1160→        if (this.gkHold <= 0) {
  1161→          // distribución automática del portero
  1162→          const mate = this.pickGkDistribution(owner)
  1163→          if (mate) {
  1164→            const dist = dist2d(owner.x, owner.z, mate.x, mate.z)
  1165→            const lofted = dist > 26
  1166→            this.doPass(owner, mate.x - owner.x, mate.z - owner.z, lofted)
  1167→            if (owner.team === 0) this.controlled = mate
  1168→          } else {
  1169→            this.kickBall(owner, this.squads[gkTeam].attackDir, 0, 16, 6)
  1170→          }
  1171→        }
  1172→        return
  1173→      }
  1174→
  1175→      // conducción: el balón "pega" al pie
  1176→      this.ownerHold += dt
  1177→      const ahead = owner.sprinting ? 0.95 : 0.55
  1178→      const tx = owner.x + Math.sin(owner.dir) * ahead
  1179→      const tz = owner.z + Math.cos(owner.dir) * ahead
  1180→      ball.vx = (tx - ball.x) * 11
  1181→      ball.vz = (tz - ball.z) * 11
  1182→      ball.vy = 0
  1183→      ball.y = BALL.r
  1184→      if (dist2d(ball.x, ball.z, owner.x, owner.z) > 2.4) this.loseControl()
  1185→      return
  1186→    }
  1187→
  1188→    // balón libre: intento de control por proximidad
  1189→    if (ball.y < PLAYER.controlHeight) {
  1190→      const all = [...this.squads[0].players, ...this.squads[1].players]
  1191→      let best: PlayerEntity | null = null
  1192→      let bestD = Infinity
  1193→      for (const p of all) {
  1194→        if (p.sentOff) continue
  1195→        if (p.kickCooldown > 0 || p.stunTimer > 0 || p.slideTimer > 0) continue
  1196→        const d = dist2d(p.x, p.z, ball.x, ball.z)
  1197→        if (d < PLAYER.reach && d < bestD) {
  1198→          bestD = d
  1199→          best = p
  1200→        }
  1201→      }
  1202→      if (best) {
  1203→        const sp = ball.speed
  1204→        const controllable = sp < 9 || Math.random() < 0.35 + best.data.skill * 0.55
  1205→        if (sp > 16 && best.isGK && this.isBallThreat(best)) {
  1206→          // atajada
  1207→          this.gainControl(best)
  1208→          this.squads[best.team].onTarget += this.pendingShot && this.shotTeam !== best.team ? 1 : 0
  1209→          if (this.pendingShot && this.shotTeam !== best.team) {
  1210→            this.addEvent(`¡Atajada de ${best.data.name}!`, 'chance')
  1211→            this.excitement = Math.max(this.excitement, 0.7)
  1212→            this.audio.ooh()
  1213→          }
  1214→          this.pendingShot = false
  1215→          this.audio.catchBall()
  1216→          return
  1217→        }
  1218→        if (controllable) {
  1219→          this.gainControl(best)
  1220→          ball.vx *= 0.15
  1221→          ball.vz *= 0.15
  1222→          ball.vy = 0
  1223→        } else {
  1224→          // control malo: rebota
  1225→          ball.vx = ball.vx * -0.25 + (Math.random() - 0.5) * 4
  1226→          ball.vz = ball.vz * -0.25 + (Math.random() - 0.5) * 4
  1227→          ball.lastTouch = best.team
  1228→          this.lastTouchPlayer = best
  1229→          best.kickCooldown = 0.25
  1230→        }
  1231→      }
  1232→    }
  1233→  }

  1261→  private checkOutOfBounds() {
  1262→    const b = this.ball
  1263→    const prevX = this.ballPrevX
  1264→
  1265→    // gol: cruza la línea de meta por debajo del travesaño y dentro de la boca
  1266→    if (prevX > 0 && b.x <= 0 && Math.abs(b.z - GOAL.zCenter) < PITCH.goalW / 2 && b.y < PITCH.goalH) {
  1267→      this.handleGoalByGoalLine(0)
  1268→      return
  1269→    }
  1270→    if (prevX < PITCH.L && b.x >= PITCH.L && Math.abs(b.z - GOAL.zCenter) < PITCH.goalW / 2 && b.y < PITCH.goalH) {
  1271→      this.handleGoalByGoalLine(PITCH.L)
  1272→      return
  1273→    }
  1274→
  1275→    // líneas de banda
  1276→    if (b.z <= 0 || b.z >= PITCH.W) {
  1277→      const team = b.lastTouch === 0 ? 1 : 0
  1278→      this.lastOutWasNearMiss = false
  1279→      if (this.pendingShot) {
  1280→        this.audio.ooh()
  1281→        this.excitement = Math.max(this.excitement, 0.5)
  1282→        this.lastOutWasNearMiss = true

  1285→      this.addEvent(this.lastOutWasNearMiss ? '¡Uy! Salió muy cerca del arco' : 'Saque de banda', 'info')
  1286→      this.pendingShot = false
  1287→      return
  1288→    }
  1289→
  1290→    // líneas de meta (sin gol)
  1291→    if (b.x <= 0 || b.x >= PITCH.L) {
  1292→      const leftSide = b.x <= 0
  1293→      const defendingTeam: TeamIndex = this.ownGoalOf(0) === (leftSide ? 0 : PITCH.L) ? 0 : 1
  1294→      const attackTeam: TeamIndex = defendingTeam === 0 ? 1 : 0
  1295→      if (b.lastTouch === defendingTeam) {
  1296→        // córner
  1297→        const z = b.z <= PITCH.W / 2 ? 0.4 : PITCH.W - 0.4
  1298→        this.setRestart({ type: 'corner', team: attackTeam, x: leftSide ? 0.4 : PITCH.L - 0.4, z })
  1299→        this.addEvent(`Córner para ${this.squads[attackTeam].roster.def.short}`, 'info')
  1300→      } else {
  1301→        // saque de puerta
  1302→        const z = clamp(GOAL.zCenter + (b.z - GOAL.zCenter) * 0.3, GOAL.zMin + 2, GOAL.zMax - 2)
  1303→        this.setRestart({
  1304→          type: 'goalkick',
  1305→          team: defendingTeam,
  1306→          x: leftSide ? PITCH.sixD : PITCH.L - PITCH.sixD,
  1307→          z,
  1308→        })
  1309→        this.addEvent('Saque de puerta', 'info')
  1310→      }
  1311→      this.pendingShot = false
  1312→    }
  1313→  }
