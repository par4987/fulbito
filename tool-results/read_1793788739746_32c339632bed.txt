   157→   157→  private placeFormation(sq: Squad, kickingTeam: TeamIndex) {
   158→   158→    for (const p of sq.players) {
   159→   159→      let wx = sq.attackDir > 0 ? p.ax * PITCH.L : (1 - p.ax) * PITCH.L
   160→   160→      let wz = p.az * PITCH.W
   161→   161→      // todos en su propia mitad en el saque
   162→   162→      if (sq.attackDir > 0) wx = Math.min(wx, PITCH.L / 2 - 0.5)
   163→   163→      else wx = Math.max(wx, PITCH.L / 2 + 0.5)
   164→   164→      // rivales fuera del círculo central
   165→   165→      if (sq.attackDir !== this.squads[kickingTeam].attackDir) {
   166→   166→        const d = dist2d(wx, wz, PITCH.L / 2, PITCH.W / 2)
   167→   167→        if (d < PITCH.circleR + 1) {
   168→   168→          const ux = (wx - PITCH.L / 2) / (d || 1)
   169→   169→          const uz = (wz - PITCH.W / 2) / (d || 1)
   170→   170→          wx = PITCH.L / 2 + ux * (PITCH.circleR + 1.5)
   171→   171→          wz = PITCH.W / 2 + uz * (PITCH.circleR + 1.5)
   172→   172→        }
   173→   173→      }
   174→   174→      p.teleport(clamp(wx, 1, PITCH.L - 1), clamp(wz, 1, PITCH.W - 1))
   175→   175→      p.targetX = p.x
   176→   176→      p.targetZ = p.z
   177→   177→    }
   178→   178→  }

   200→   200→      if (info.team === 0) this.controlled = best
   201→   201→      else this.controlled = null
   202→   202→    }

   290→   290→  /** Pase raso o alto hacia el mejor compañero según dirección deseada. */
   291→   291→  doPass(p: PlayerEntity, aimX: number, aimZ: number, lofted = false) {
   292→   292→    if (p.kickCooldown > 0) return
   293→   293→    const receiver = this.pickPassTarget(p, aimX, aimZ, lofted)
   294→   294→    if (!receiver) {
   295→   295→      // despeje hacia adelante
   296→   296→      const dir = this.squads[p.team].attackDir
   297→   297→      this.kickBall(p, dir, (Math.random() - 0.5) * 20, 16, 4)
   298→   298→      return
   299→   299→    }
   300→   300→    const err = (1 - p.data.skill) * this.diff.passErr * (lofted ? 1.6 : 1)
   301→   301→    const noise = (Math.random() - 0.5) * 2 * err
   302→   302→    const cs = Math.cos(noise)
   303→   303→    const sn = Math.sin(noise)
   304→   304→    const dist = dist2d(p.x, p.z, receiver.x, receiver.z)
   305→   305→    const speed = lofted ? clamp(10 + dist * 0.42, 13, 20) : clamp(9 + dist * 0.55, 10, 21)
   306→   306→    // punto adelantado hacia donde corre el receptor
   307→   307→    const lead = lofted ? 0.35 : 0.45
   308→   308→    let dx = receiver.x + receiver.vx * lead - p.x
   309→   309→    let dz = receiver.z + receiver.vz * lead - p.z
   310→   310→    const rx = dx * cs - dz * sn
   311→   311→    const rz = dx * sn + dz * cs
   312→   312→    const vz = lofted ? clamp(dist * 0.32, 3, 8.5) : 0.35
   313→   313→    this.kickBall(p, rx, rz, speed, vz)
   314→   314→    if (p.team === 0 && !lofted) this.controlled = receiver
   315→   315→    if (p.team === 1) this.aiPassHint = receiver
   316→   316→  }
   317→   317→
   318→   318→  private pickPassTarget(p: PlayerEntity, aimX: number, aimZ: number, lofted: boolean): PlayerEntity | null {
   319→   319→    const aimLen = Math.hypot(aimX, aimZ)
   320→   320→    const nx = aimLen > 0.1 ? aimX / aimLen : Math.sin(p.dir)
   321→   321→    const nz = aimLen > 0.1 ? aimZ / aimLen : Math.cos(p.dir)
   322→   322→    const squad = this.squads[p.team]
   323→   323→    let best: PlayerEntity | null = null
   324→   324→    let bestScore = -Infinity
   325→   325→    for (const mate of squad.players) {
   326→   326→      if (mate === p || mate.isGK) continue
   327→   327→      const dx = mate.x - p.x
   328→   328→      const dz = mate.z - p.z
   329→   329→      const d = Math.hypot(dx, dz)
   330→   330→      if (d < 2.5 || d > (lofted ? 55 : 34)) continue
   331→   331→      const dot = (dx / d) * nx + (dz / d) * nz
   332→   332→      if (!lofted && dot < 0.2) continue
   333→   333→      const open = this.closestOppDist(mate)
   334→   334→      let lanePen = 0
   335→   335→      if (!lofted) lanePen = this.laneBlocked(p, mate) ? 2.5 : 0
   336→   336→      const score = dot * 3 + open * 0.35 - d * 0.045 - lanePen
   337→   337→      if (score > bestScore) {
   338→   338→        bestScore = score
   339→   339→        best = mate
   340→   340→      }
   341→   341→    }
   342→   342→    return best
   343→   343→  }
   344→   344→
   345→   345→  private closestOppDist(p: PlayerEntity): number {
   346→   346→    const opp = this.squads[p.team === 0 ? 1 : 0].players
   347→   347→    let d = Infinity
   348→   348→    for (const o of opp) d = Math.min(d, dist2d(o.x, o.z, p.x, p.z))
   349→   349→    return d
   350→   350→  }
   351→   351→
   352→   352→  private laneBlocked(from: PlayerEntity, to: PlayerEntity): boolean {
   353→   353→    const opp = this.squads[from.team === 0 ? 1 : 0].players
   354→   354→    const dx = to.x - from.x
   355→   355→    const dz = to.z - from.z
   356→   356→    const len = Math.hypot(dx, dz) || 1
   357→   357→    for (const o of opp) {
   358→   358→      const t = clamp(((o.x - from.x) * dx + (o.z - from.z) * dz) / (len * len), 0.1, 0.9)
   359→   359→      const px = from.x + dx * t
   360→   360→      const pz = from.z + dz * t
   361→   361→      if (dist2d(o.x, o.z, px, pz) < 1.3) return true
   362→   362→    }
   363→   363→    return false
   364→   364→  }

   366→   366→  /** Disparo al arco. power 0..1, aim = componente perpendicular del input. */
   367→   367→  doShoot(p: PlayerEntity, power: number, aim: number) {
   368→   368→    if (p.kickCooldown > 0) return
   369→   369→    const gx = this.goalOf(p.team)
   370→   370→    const dirX = gx > p.x ? 1 : -1
   371→   371→    const dist = Math.abs(gx - p.x)
   372→   372→    const noise = (1 - p.data.skill) * this.diff.shootErr * (2 + dist * 0.06) + power * 0.04
   373→   373→    const angle = (Math.random() - 0.5) * 2 * noise
   374→   374→    let dz = GOAL.zCenter - p.z + aim * 3.4
   375→   375→    let dx = dirX * Math.max(8, dist)
   376→   376→    const cs = Math.cos(angle)
   377→   377→    const sn = Math.sin(angle)
   378→   378→    const rx = dx * cs - dz * sn
   379→   379→    const rz = dx * sn + dz * cs
   380→   380→    const speed = 18 + power * 12
   381→   381→    const vz = 0.5 + power * 3.1
   382→   382→    this.kickBall(p, rx, rz, speed, vz)
   383→   383→    this.squads[p.team].shots++
   384→   384→    this.pendingShot = true
   385→   385→    this.shotTeam = p.team
   386→   386→    this.shotWindow = 1.3
   387→   387→  }

   389→   389→  /** Entrada (presión): intenta robar el balón al dueño cercano. */
   390→   390→  tryTackle(p: PlayerEntity) {
   391→   391→    if (p.tackleCd > 0 || p.stunTimer > 0) return
   392→   392→    p.tackleCd = PLAYER.tackleCooldown
   393→   393→    const owner = this.owner
   394→   394→    if (owner && owner.team !== p.team && dist2d(p.x, p.z, owner.x, owner.z) < PLAYER.tackleRange) {
   395→   395→      const prob = 0.45 + p.data.skill * 0.25 + this.diff.tackle * 0.15 - owner.data.skill * 0.2
   396→   396→      if (Math.random() < prob) {
   397→   397→        this.loseControl()
   398→   398→        this.ball.kick(p.vx + Math.sin(p.dir) * 4, p.vz + Math.cos(p.dir) * 4, 7, 0.4)
   399→   399→        this.ball.lastTouch = p.team
   400→   400→        this.lastTouchPlayer = p
   401→   401→        owner.stunTimer = 0.3
   402→   402→        this.audio.kick(0.3)
   403→   403→      } else {
   404→   404→        p.stunTimer = 0.35
   405→   405→      }
   406→   406→    } else if (!owner && dist2d(p.x, p.z, this.ball.x, this.ball.z) < PLAYER.tackleRange) {
   407→   407→      // empujón al balón libre
   408→   408→      this.ball.kick(Math.sin(p.dir), Math.cos(p.dir), 8, 0.5)
   409→   409→      this.ball.lastTouch = p.team
   410→   410→      this.lastTouchPlayer = p
   411→   411→      p.kickCooldown = 0.3
   412→   412→    }
   413→   413→  }

   421→   421→  update(dt: number) {
   422→   422→    this.excitement = Math.max(0, this.excitement - dt * 0.35)
   423→   423→    this.audio.setExcitement(this.excitement * 0.8)
   424→   424→    this.shotWindow = Math.max(0, this.shotWindow - dt)
   425→   425→    if (this.shotWindow === 0) this.pendingShot = false
   426→   426→
   427→   427→    switch (this.phase) {
   428→   428→      case 'kickoff':
   429→   429→      case 'restart':
   430→   430→      case 'half': {
   431→   431→        this.phaseTimer -= dt
   432→   432→        if (this.phase === 'half') {
   433→   433→          if (this.phaseTimer <= 0) {
   434→   434→            this.half = 2
   435→   435→            for (const sq of this.squads) sq.attackDir = (sq.attackDir * -1) as 1 | -1
   436→   436→            this.setupKickoff(1)
   437→   437→            this.addEvent('Comienza el segundo tiempo', 'info')
   438→   438→          }
   439→   439→          break
   440→   440→        }
   441→   441→        // congelados: solo animación
   442→   442→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   443→   443→        this.ballPrevX = this.ball.x
   444→   444→        if (this.phaseTimer <= 0) {
   445→   445→          this.phase = 'play'
   446→   446→        }
   447→   447→        break
   448→   448→      }
   449→   449→      case 'goal': {
   450→   450→        this.phaseTimer -= dt
   451→   451→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   452→   452→        if (this.phaseTimer <= 0) {
   453→   453→          this.celebrateScorer = null
   454→   454→          this.celebrateTeam = null
   455→   455→          this.setupKickoff(this.lastTouchTeamConceding())
   456→   456→        }
   457→   457→        break
   458→   458→      }
   459→   459→      case 'play':
   460→   460→        this.updatePlay(dt)
   461→   461→        break
   462→   462→      case 'full':
   463→   463→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   464→   464→        break
   465→   465→    }
   466→   466→    this.input.endFrame()
   467→   467→  }

   469→   469→  private lastTouchTeamConceding(): TeamIndex {
   470→   470→    if (!this.lastTouchPlayer) return this.kickoffTeam === 0 ? 1 : 0
   471→   471→    // el equipo que recibió el gol saca
   472→   472→    return this.lastTouchPlayer.team === 0 ? 1 : 0
   473→   473→  }

   475→   475→  private updatePlay(dt: number) {
   476→   476→    this.clock += dt
   477→   477→    this.ballPrevX = this.ball.x
   478→   478→
   479→   479→    this.updateControlledInput(dt)
   480→   480→    updateMatchAI(this, dt)
   481→   481→    for (const sq of this.squads) {
   482→   482→      for (const p of sq.players) p.update(dt)
   483→   483→    }
   484→   484→    this.separatePlayers()
   485→   485→    this.updateBallPossession(dt)
   486→   486→    this.checkOutOfBounds()
   487→   487→
   488→   488→    if (this.owner) this.squads[this.owner.team].possession += dt
   489→   489→
   490→   490→    if (this.half === 1 && this.clock >= this.durationSec / 2) {
   491→   491→      this.finishHalf()
   492→   492→    } else if (this.half === 2 && this.clock >= this.durationSec && !this.finished) {
   493→   493→      this.goFullTime()
   494→   494→    }
   495→   495→  }

   497→   497→  /** Convierte la entrada humana en acciones del jugador controlado. */
   498→   498→  private updateControlledInput(dt: number) {
   499→   499→    const inp = this.input
   500→   500→    // auto-cambio de jugador
   501→   501→    const team0 = this.squads[0]
   502→   502→    if (!this.owner || this.owner.team !== 0) {
   503→   503→      this.switchHysteresis += dt
   504→   504→      if (this.switchHysteresis > 0.18 || !this.controlled || this.controlled.team !== 0) {
   505→   505→        const fut = this.ball.future(0.25)
   506→   506→        let best: PlayerEntity | null = null
   507→   507→        let bestD = Infinity
   508→   508→        const nearOwnBox = Math.abs(this.ball.x - this.ownGoalOf(0)) < 20
   509→   509→        for (const p of team0.players) {
   510→   510→          if (p.isGK && !nearOwnBox) continue
   511→   511→          const d = dist2d(p.x, p.z, fut.x, fut.z)
   512→   512→          if (d < bestD) {
   513→   513→            bestD = d
   514→   514→            best = p
   515→   515→          }
   516→   516→        }
   517→   517→        if (best && best !== this.controlled) this.controlled = best
   518→   518→        this.switchHysteresis = 0
   519→   519→      }
   520→   520→    } else {
   521→   521→      this.switchHysteresis = 0
   522→   522→      this.controlled = this.owner
   523→   523→    }
   524→   524→
   525→   525→    const c = this.controlled
   526→   526→    if (!c) return
   527→   527→    c.speedMult = this.diff.userBoost
   528→   528→
   529→   529→    // movimiento relativo a cámara (el engine lo traduce antes: aquí ya es mundo)
   530→   530→    const mx = inp.move.x
   531→   531→    const my = inp.move.y
   532→   532→    const world = this.engineMoveToWorld(mx, my)
   533→   533→    const moving = Math.hypot(world.x, world.z) > 0.1
   534→   534→    c.sprinting = inp.sprint && moving
   535→   535→    if (moving) {
   536→   536→      const len = Math.hypot(world.x, world.z)
   537→   537→      c.desiredVx = (world.x / len) * c.maxSpeed()
   538→   538→      c.desiredVz = (world.z / len) * c.maxSpeed()
   539→   539→    } else {
   540→   540→      c.desiredVx = 0
   541→   541→      c.desiredVz = 0
   542→   542→    }
   543→   543→
   544→   544→    const isOwner = this.owner === c
   545→   545→    if (isOwner) {
   546→   546→      // disparo con carga
   547→   547→      if (inp.isHeld('shoot')) {
   548→   548→        this.charging = true
   549→   549→        this.charge = Math.min(1, this.charge + dt / 0.85)
   550→   550→        if (this.charge >= 1) {
   551→   551→          this.releaseShot(c)
   552→   552→        }
   553→   553→      } else if (this.charging) {
   554→   554→        this.releaseShot(c)
   555→   555→      }
   556→   556→      if (inp.consumePressed('pass')) {
   557→   557→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   558→   558→        this.doPass(c, aim.x, aim.z, false)
   559→   559→        this.charge = 0
   560→   560→        this.charging = false
   561→   561→      }
   562→   562→      if (inp.consumePressed('lob')) {
   563→   563→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   564→   564→        this.doPass(c, aim.x, aim.z, true)
   565→   565→      }
   566→   566→    } else {
   567→   567→      // defensa
   568→   568→      this.charge = 0
   569→   569→      this.charging = false
   570→   570→      if (inp.consumePressed('pass')) this.tryTackle(c)
   571→   571→      if (inp.consumePressed('shoot')) {
   572→   572→        const dx = moving ? world.x : Math.sin(c.dir)
   573→   573→        const dz = moving ? world.z : Math.cos(c.dir)
   574→   574→        this.slideTackle(c, dx, dz)
   575→   575→      }
   576→   576→      if (inp.consumePressed('lob') || inp.consumePressed('switch')) {
   577→   577→        // cambio manual: siguiente más cercano
   578→   578→        const fut = this.ball.future(0.25)
   579→   579→        const others = team0.players
   580→   580→          .filter((p) => !p.isGK && p !== c)
   581→   581→          .sort((a, b) => dist2d(a.x, a.z, fut.x, fut.z) - dist2d(b.x, b.z, fut.x, fut.z))
   582→   582→        if (others[0]) this.controlled = others[0]
   583→   583→      }
   584→   584→    }
   585→   585→  }

   587→   587→  private releaseShot(c: PlayerEntity) {
   588→   588→    const world = this.engineMoveToWorld(this.input.move.x, this.input.move.y)
   589→   589→    // input vertical en mundo: arriba = palo lejano, abajo = palo cercano
   590→   590→    const aimProj = clamp(world.z, -1, 1)
   591→   591→    this.doShoot(c, Math.max(0.25, this.charge), aimProj)
   592→   592→    this.charge = 0
   593→   593→    this.charging = false
   594→   594→  }

   597→   597→  cameraMode: 'broadcast' | 'aerea' | 'arco' = 'broadcast'

   599→   599→  private engineMoveToWorld(x: number, y: number): { x: number; z: number } {
   600→   600→    if (this.cameraMode === 'arco') {
   601→   601→      // cámara tras el arco que ataca el usuario
   602→   602→      const f = this.attackDirUser

   610→   610→  private separatePlayers() {
   611→   611→    const all = [...this.squads[0].players, ...this.squads[1].players]
   612→   612→    for (let i = 0; i < all.length; i++) {
   613→   613→      for (let j = i + 1; j < all.length; j++) {
   614→   614→        const a = all[i]
   615→   615→        const b = all[j]
   616→   616→        const dx = b.x - a.x
   617→   617→        const dz = b.z - a.z
   618→   618→        const d = Math.hypot(dx, dz)
   619→   619→        const min = PLAYER.radius * 2
   620→   620→        if (d > 0.001 && d < min) {
   621→   621→          const push = (min - d) / 2
   622→   622→          const ux = dx / d
   623→   623→          const uz = dz / d
   624→   624→          a.x -= ux * push
   625→   625→          a.z -= uz * push
   626→   626→          b.x += ux * push
   627→   627→          b.z += uz * push
   628→   628→        }
   629→   629→      }
   630→   630→    }

   633→   633→  private updateBallPossession(dt: number) {
   634→   634→    const ball = this.ball
   635→   635→    const postHit = ball.update(dt)
   636→   636→    if (postHit === 'post') {
   637→   637→      this.audio.post()
   638→   638→      this.addEvent('¡Al palo!', 'chance')
   639→   639→      this.excitement = Math.max(this.excitement, 0.6)
   640→   640→    } else if (postHit === 'crossbar') {
   641→   641→      this.audio.post()
   642→   642→      this.addEvent('¡Al travesaño!', 'chance')
   643→   643→      this.excitement = Math.max(this.excitement, 0.6)
   644→   644→    }
   645→   645→
   646→   646→    const owner = this.owner
   647→   647→    if (owner) {
   648→   648→      if (owner.isGK && this.gkHold > 0) {
   649→   649→        this.gkHold -= dt
   650→   650→        // balón en las manos
   651→   651→        ball.x = owner.x + Math.sin(owner.dir) * 0.35
   652→   652→        ball.z = owner.z + Math.cos(owner.dir) * 0.35
   653→   653→        ball.y = 1.05
   654→   654→        ball.stop()
   655→   655→        const gkTeam = owner.team
   656→   656→        if (this.gkHold <= 0) {
   657→   657→          // distribución automática del portero
   658→   658→          const mate = this.pickGkDistribution(owner)
   659→   659→          if (mate) {
   660→   660→            const dist = dist2d(owner.x, owner.z, mate.x, mate.z)
   661→   661→            const lofted = dist > 26
   662→   662→            this.doPass(owner, mate.x - owner.x, mate.z - owner.z, lofted)
   663→   663→            if (owner.team === 0) this.controlled = mate
   664→   664→          } else {
   665→   665→            this.kickBall(owner, this.squads[gkTeam].attackDir, 0, 16, 6)
   666→   666→          }
   667→   667→        }
   668→   668→        return
   669→   669→      }
   670→   670→
   671→   671→      // conducción: el balón "pega" al pie
   672→   672→      this.ownerHold += dt
   673→   673→      const ahead = owner.sprinting ? 0.95 : 0.55
   674→   674→      const tx = owner.x + Math.sin(owner.dir) * ahead
   675→   675→      const tz = owner.z + Math.cos(owner.dir) * ahead
   676→   676→      ball.vx = (tx - ball.x) * 11
   677→   677→      ball.vz = (tz - ball.z) * 11
   678→   678→      ball.vy = 0
   679→   679→      ball.y = BALL.r
   680→   680→      if (dist2d(ball.x, ball.z, owner.x, owner.z) > 2.4) this.loseControl()
   681→   681→      return
   682→   682→    }
   683→   683→
   684→   684→    // balón libre: intento de control por proximidad
   685→   685→    if (ball.y < PLAYER.controlHeight) {
   686→   686→      const all = [...this.squads[0].players, ...this.squads[1].players]
   687→   687→      let best: PlayerEntity | null = null
   688→   688→      let bestD = Infinity
   689→   689→      for (const p of all) {
   690→   690→        if (p.kickCooldown > 0 || p.stunTimer > 0 || p.slideTimer > 0) continue
   691→   691→        const d = dist2d(p.x, p.z, ball.x, ball.z)
   692→   692→        if (d < PLAYER.reach && d < bestD) {
   693→   693→          bestD = d
   694→   694→          best = p
   695→   695→        }
   696→   696→      }
   697→   697→      if (best) {
   698→   698→        const sp = ball.speed
   699→   699→        const controllable = sp < 9 || Math.random() < 0.35 + best.data.skill * 0.55
   700→   700→        if (sp > 16 && best.isGK && this.isBallThreat(best)) {
   701→   701→          // atajada
   702→   702→          this.gainControl(best)
   703→   703→          this.squads[best.team].onTarget += this.pendingShot && this.shotTeam !== best.team ? 1 : 0
   704→   704→          if (this.pendingShot && this.shotTeam !== best.team) {
   705→   705→            this.addEvent(`¡Atajada de ${best.data.name}!`, 'chance')
   706→   706→            this.excitement = Math.max(this.excitement, 0.7)
   707→   707→            this.audio.ooh()
   708→   708→          }
   709→   709→          this.pendingShot = false
   710→   710→          this.audio.catchBall()
   711→   711→          return
   712→   712→        }
   713→   713→        if (controllable) {
   714→   714→          this.gainControl(best)
   715→   715→          ball.vx *= 0.15
   716→   716→          ball.vz *= 0.15
   717→   717→          ball.vy = 0
   718→   718→        } else {
   719→   719→          // control malo: rebota
   720→   720→          ball.vx = ball.vx * -0.25 + (Math.random() - 0.5) * 4
   721→   721→          ball.vz = ball.vz * -0.25 + (Math.random() - 0.5) * 4
   722→   722→          ball.lastTouch = best.team
   723→   723→          this.lastTouchPlayer = best
   724→   724→          best.kickCooldown = 0.25
   725→   725→        }
   726→   726→      }
   727→   727→    }
   728→   728→  }

   730→   730→  private isBallThreat(gk: PlayerEntity): boolean {
   731→   731→    // el balón va hacia su arco rápido
   732→   732→    const gx = this.ownGoalOf(gk.team)
   733→   733→    const toward = gx < PITCH.L / 2 ? this.ball.vx < -6 : this.ball.vx > 6

   737→   737→  private pickGkDistribution(gk: PlayerEntity): PlayerEntity | null {
   738→   738→    const squad = this.squads[gk.team]
   739→   739→    let best: PlayerEntity | null = null
   740→   740→    let bestScore = -Infinity
   741→   741→    for (const mate of squad.players) {
   742→   742→      if (mate === gk) continue
   743→   743→      const d = dist2d(gk.x, gk.z, mate.x, mate.z)
   744→   744→      const open = Math.min(9, this.closestOppDist(mate))
   745→   745→      const score = open * 2 - d * 0.06 + (mate.data.role === 'DF' ? 2 : 0)
   746→   746→      if (score > bestScore) {
   747→   747→        bestScore = score
   748→   748→        best = mate
   749→   749→      }
   750→   750→    }
   751→   751→    return best
   752→   752→  }

   756→   756→  private checkOutOfBounds() {
   757→   757→    const b = this.ball
   758→   758→    const prevX = this.ballPrevX
   759→   759→
   760→   760→    // gol: cruza la línea de meta por debajo del travesaño y dentro de la boca
   761→   761→    if (prevX > 0 && b.x <= 0 && Math.abs(b.z - GOAL.zCenter) < PITCH.goalW / 2 && b.y < PITCH.goalH) {
   762→   762→      this.handleGoalByGoalLine(0)
   763→   763→      return
   764→   764→    }
   765→   765→    if (prevX < PITCH.L && b.x >= PITCH.L && Math.abs(b.z - GOAL.zCenter) < PITCH.goalW / 2 && b.y < PITCH.goalH) {
   766→   766→      this.handleGoalByGoalLine(PITCH.L)
   767→   767→      return
   768→   768→    }
   769→   769→
   770→   770→    // líneas de banda
   771→   771→    if (b.z <= 0 || b.z >= PITCH.W) {
   772→   772→      const team = b.lastTouch === 0 ? 1 : 0
   773→   773→      this.lastOutWasNearMiss = false
   774→   774→      if (this.pendingShot) {
   775→   775→        this.audio.ooh()
   776→   776→        this.excitement = Math.max(this.excitement, 0.5)
   777→   777→        this.lastOutWasNearMiss = true
   778→   778→      }
   779→   779→      this.setRestart({ type: 'throwin', team, x: clamp(b.x, 1.5, PITCH.L - 1.5), z: b.z <= 0 ? 0.3 : PITCH.W - 0.3 })
   780→   780→      this.addEvent(this.lastOutWasNearMiss ? '¡Uy! Salió muy cerca del arco' : 'Saque de banda', 'info')
   781→   781→      this.pendingShot = false
   782→   782→      return
   783→   783→    }
   784→   784→
   785→   785→    // líneas de meta (sin gol)
   786→   786→    if (b.x <= 0 || b.x >= PITCH.L) {
   787→   787→      const leftSide = b.x <= 0
   788→   788→      const defendingTeam: TeamIndex = this.ownGoalOf(0) === (leftSide ? 0 : PITCH.L) ? 0 : 1
   789→   789→      const attackTeam: TeamIndex = defendingTeam === 0 ? 1 : 0
   790→   790→      if (b.lastTouch === defendingTeam) {
   791→   791→        // córner
   792→   792→        const z = b.z <= PITCH.W / 2 ? 0.4 : PITCH.W - 0.4
   793→   793→        this.setRestart({ type: 'corner', team: attackTeam, x: leftSide ? 0.4 : PITCH.L - 0.4, z })
   794→   794→        this.addEvent(`Córner para ${this.squads[attackTeam].roster.def.short}`, 'info')
   795→   795→      } else {
   796→   796→        // saque de puerta
   797→   797→        const z = clamp(GOAL.zCenter + (b.z - GOAL.zCenter) * 0.3, GOAL.zMin + 2, GOAL.zMax - 2)
   798→   798→        this.setRestart({
   799→   799→          type: 'goalkick',
   800→   800→          team: defendingTeam,
   801→   801→          x: leftSide ? PITCH.sixD : PITCH.L - PITCH.sixD,
   802→   802→          z,
   803→   803→        })
   804→   804→        this.addEvent('Saque de puerta', 'info')
   805→   805→      }
   806→   806→      this.pendingShot = false
   807→   807→    }
   808→   808→  }
