   113→  private buildSquad(roster: TeamRoster, team: TeamIndex, attackDir: 1 | -1): Squad {

   165→  setupKickoff(kickingTeam: TeamIndex) {
   166→    this.phase = 'kickoff'
   167→    this.phaseTimer = MATCH_RULES.kickoffFreeze
   168→    this.kickoffTeam = kickingTeam
   169→    this.owner = null
   170→    this.ball.reset()
   171→    for (const sq of this.squads) this.placeFormation(sq, kickingTeam)
   172→    const taker = this.squads[kickingTeam].players[9] // FW central
   173→    taker.teleport(PITCH.L / 2 - this.squads[kickingTeam].attackDir * 0.8, PITCH.W / 2)
   174→    this.gainControl(taker)
   175→    this.controlled = kickingTeam === 0 ? taker : null
   176→    this.audio.whistle(1)

   179→  private placeFormation(sq: Squad, kickingTeam: TeamIndex) {
   180→    for (const p of sq.players) {
   181→      let wx = sq.attackDir > 0 ? p.ax * PITCH.L : (1 - p.ax) * PITCH.L
   182→      let wz = p.az * PITCH.W
   183→      // todos en su propia mitad en el saque
   184→      if (sq.attackDir > 0) wx = Math.min(wx, PITCH.L / 2 - 0.5)
   185→      else wx = Math.max(wx, PITCH.L / 2 + 0.5)
   186→      // rivales fuera del círculo central
   187→      if (sq.attackDir !== this.squads[kickingTeam].attackDir) {
   188→        const d = dist2d(wx, wz, PITCH.L / 2, PITCH.W / 2)
   189→        if (d < PITCH.circleR + 1) {
   190→          const ux = (wx - PITCH.L / 2) / (d || 1)
   191→          const uz = (wz - PITCH.W / 2) / (d || 1)
   192→          wx = PITCH.L / 2 + ux * (PITCH.circleR + 1.5)
   193→          wz = PITCH.W / 2 + uz * (PITCH.circleR + 1.5)
   194→        }
   195→      }
   196→      p.teleport(clamp(wx, 1, PITCH.L - 1), clamp(wz, 1, PITCH.W - 1))
   197→      p.targetX = p.x
   198→      p.targetZ = p.z
   199→    }
   200→  }

   202→  private setRestart(info: RestartInfo) {
   203→    this.phase = 'restart'
   204→    this.phaseTimer = MATCH_RULES.restartFreeze
   205→    this.restart = info
   206→    this.ball.reset(info.x, info.z)
   207→    this.owner = null
   208→    const squad = this.squads[info.team]
   209→    let best: PlayerEntity | null = null
   210→    let bestD = Infinity
   211→    for (const p of squad.players) {
   212→      if (info.type === 'goalkick' ? !p.isGK : p.isGK) continue
   213→      const d = dist2d(p.x, p.z, info.x, info.z)
   214→      if (d < bestD) {
   215→        bestD = d
   216→        best = p
   217→      }
   218→    }
   219→    if (best) {
   220→      best.teleport(info.x - Math.sign(info.x - PITCH.L / 2) * 0.9, info.z)
   221→      this.gainControl(best)
   222→      if (info.team === 0) this.controlled = best
   223→      else this.controlled = null
   224→    }
   225→  }

   227→  private handleGoal(scoringTeam: TeamIndex) {
   228→    const sq = this.squads[scoringTeam]
   229→    sq.score++
   230→    if (this.pendingShot && this.shotTeam === scoringTeam) {
   231→      sq.onTarget++
   232→    }
   233→    const scorer = this.lastTouchPlayer
   234→    const own = scorer && scorer.team !== scoringTeam
   235→    const name = scorer ? scorer.data.name : 'desconocido'
   236→    this.celebrateScorer = own ? `${name} (autogol)` : name
   237→    this.celebrateTeam = scoringTeam
   238→    this.phase = 'goal'
   239→    this.phaseTimer = MATCH_RULES.goalCelebration
   240→    this.owner = null
   241→    this.pendingShot = false
   242→    this.audio.goalRoar()
   243→    this.excitement = 1
   244→    this.addEvent(`¡GOL de ${this.celebrateScorer} (${sq.roster.def.short})!`, 'goal')
   245→  }

   301→  private kickBall(p: PlayerEntity, dx: number, dz: number, speed: number, vz: number) {
   302→    this.ball.kick(dx, dz, speed, vz)

   312→  /** Pase raso o alto hacia el mejor compañero según dirección deseada. */
   313→  doPass(p: PlayerEntity, aimX: number, aimZ: number, lofted = false) {
   314→    if (p.kickCooldown > 0) return
   315→    const receiver = this.pickPassTarget(p, aimX, aimZ, lofted)
   316→    if (!receiver) {
   317→      // despeje hacia adelante
   318→      const dir = this.squads[p.team].attackDir
   319→      this.kickBall(p, dir, (Math.random() - 0.5) * 20, 16, 4)
   320→      return
   321→    }
   322→    const err = (1 - p.data.skill) * this.diff.passErr * (lofted ? 1.6 : 1)
   323→    const noise = (Math.random() - 0.5) * 2 * err
   324→    const cs = Math.cos(noise)
   325→    const sn = Math.sin(noise)
   326→    const dist = dist2d(p.x, p.z, receiver.x, receiver.z)
   327→    const speed = lofted ? clamp(10 + dist * 0.42, 13, 20) : clamp(9 + dist * 0.55, 10, 21)
   328→    // punto adelantado hacia donde corre el receptor
   329→    const lead = lofted ? 0.35 : 0.45
   330→    let dx = receiver.x + receiver.vx * lead - p.x
   331→    let dz = receiver.z + receiver.vz * lead - p.z
   332→    const rx = dx * cs - dz * sn
   333→    const rz = dx * sn + dz * cs
   334→    const vz = lofted ? clamp(dist * 0.32, 3, 8.5) : 0.35
   335→    this.kickBall(p, rx, rz, speed, vz)
   336→    if (p.team === 0 && !lofted) this.controlled = receiver
   337→    if (p.team === 1) this.aiPassHint = receiver
   338→  }
   339→
   340→  private pickPassTarget(p: PlayerEntity, aimX: number, aimZ: number, lofted: boolean): PlayerEntity | null {
   341→    const aimLen = Math.hypot(aimX, aimZ)
   342→    const nx = aimLen > 0.1 ? aimX / aimLen : Math.sin(p.dir)
   343→    const nz = aimLen > 0.1 ? aimZ / aimLen : Math.cos(p.dir)
   344→    const squad = this.squads[p.team]
   345→    let best: PlayerEntity | null = null
   346→    let bestScore = -Infinity
   347→    for (const mate of squad.players) {
   348→      if (mate === p || mate.isGK) continue
   349→      const dx = mate.x - p.x
   350→      const dz = mate.z - p.z
   351→      const d = Math.hypot(dx, dz)
   352→      if (d < 2.5 || d > (lofted ? 55 : 34)) continue
   353→      const dot = (dx / d) * nx + (dz / d) * nz
   354→      if (!lofted && dot < 0.2) continue
   355→      const open = this.closestOppDist(mate)
   356→      let lanePen = 0
   357→      if (!lofted) lanePen = this.laneBlocked(p, mate) ? 2.5 : 0
   358→      const score = dot * 3 + open * 0.35 - d * 0.045 - lanePen
   359→      if (score > bestScore) {
   360→        bestScore = score
   361→        best = mate
   362→      }
   363→    }
   364→    return best
   365→  }

   367→  private closestOppDist(p: PlayerEntity): number {
   368→    const opp = this.squads[p.team === 0 ? 1 : 0].players
   369→    let d = Infinity
   370→    for (const o of opp) d = Math.min(d, dist2d(o.x, o.z, p.x, p.z))
   371→    return d
   372→  }
   373→
   374→  private laneBlocked(from: PlayerEntity, to: PlayerEntity): boolean {
   375→    const opp = this.squads[from.team === 0 ? 1 : 0].players
   376→    const dx = to.x - from.x
   377→    const dz = to.z - from.z
   378→    const len = Math.hypot(dx, dz) || 1
   379→    for (const o of opp) {
   380→      const t = clamp(((o.x - from.x) * dx + (o.z - from.z) * dz) / (len * len), 0.1, 0.9)
   381→      const px = from.x + dx * t
   382→      const pz = from.z + dz * t
   383→      if (dist2d(o.x, o.z, px, pz) < 1.3) return true
   384→    }
   385→    return false
   386→  }

   388→  /** Disparo al arco. power 0..1, aim = componente perpendicular del input. */
   389→  doShoot(p: PlayerEntity, power: number, aim: number) {
   390→    if (p.kickCooldown > 0) return
   391→    const gx = this.goalOf(p.team)
   392→    const dirX = gx > p.x ? 1 : -1
   393→    const dist = Math.abs(gx - p.x)
   394→    const noise = (1 - p.data.skill) * this.diff.shootErr * (2 + dist * 0.06) + power * 0.04
   395→    const angle = (Math.random() - 0.5) * 2 * noise
   396→    let dz = GOAL.zCenter - p.z + aim * 3.4
   397→    let dx = dirX * Math.max(8, dist)
   398→    const cs = Math.cos(angle)
   399→    const sn = Math.sin(angle)
   400→    const rx = dx * cs - dz * sn
   401→    const rz = dx * sn + dz * cs
   402→    const speed = 18 + power * 12
   403→    const vz = 0.5 + power * 3.1
   404→    this.kickBall(p, rx, rz, speed, vz)
   405→    this.squads[p.team].shots++
   406→    this.pendingShot = true
   407→    this.shotTeam = p.team
   408→    this.shotWindow = 1.3
   409→  }

   411→  /** Entrada (presión): intenta robar el balón al dueño cercano. */
   412→  tryTackle(p: PlayerEntity) {
   413→    if (p.tackleCd > 0 || p.stunTimer > 0) return
   414→    p.tackleCd = PLAYER.tackleCooldown
   415→    const owner = this.owner
   416→    if (owner && owner.team !== p.team && dist2d(p.x, p.z, owner.x, owner.z) < PLAYER.tackleRange) {
   417→      const prob = 0.45 + p.data.skill * 0.25 + this.diff.tackle * 0.15 - owner.data.skill * 0.2
   418→      if (Math.random() < prob) {
   419→        this.loseControl()
   420→        this.ball.kick(p.vx + Math.sin(p.dir) * 4, p.vz + Math.cos(p.dir) * 4, 7, 0.4)
   421→        this.ball.lastTouch = p.team
   422→        this.lastTouchPlayer = p
   423→        owner.stunTimer = 0.3
   424→        this.audio.kick(0.3)
   425→      } else {
   426→        p.stunTimer = 0.35
   427→      }
   428→    } else if (!owner && dist2d(p.x, p.z, this.ball.x, this.ball.z) < PLAYER.tackleRange) {
   429→      // empujón al balón libre
   430→      this.ball.kick(Math.sin(p.dir), Math.cos(p.dir), 8, 0.5)
   431→      this.ball.lastTouch = p.team
   432→      this.lastTouchPlayer = p
   433→      p.kickCooldown = 0.3
   434→    }
   435→  }

   467→  /**
   468→   * Realiza un cambio: el titular conserva su puesto en la formación y el rig 3D;
   469→   * solo se rebrandea (nombre, dorsal, atributos) y recupera resistencia.
   470→   * Portero solo por portero (el rig del ARQ es distinto y su IA es especial).
   471→   */
   472→  makeSubstitution(team: TeamIndex, outIdx: number, benchIdx: number): SubResult {
   473→    if (this.phase === 'full') return 'invalido'
   474→    const sq = this.squads[team]
   475→    if (sq.subsUsed >= MATCH_RULES.maxSubs) return 'limite'
   476→    const out = sq.players[outIdx]
   477→    const sub = sq.bench[benchIdx]
   478→    if (!out || !sub || sq.benchUsed[benchIdx]) return 'invalido'
   479→    if (out.isGK !== (sub.role === 'GK')) return 'invalido'
   480→    const oldName = out.data.name
   481→    out.data = sub
   482→    out.stamina = 100
   483→    sq.benchUsed[benchIdx] = true
   484→    sq.subsUsed++
   485→    this.addEvent(`Cambio ${sq.roster.def.short}: entra ${sub.name} por ${oldName}`, 'info')
   486→    return 'ok'
   487→  }

   491→  /** La CPU reemplaza a su jugador de campo más cansado a partir del minuto 55. */
   492→  private tryCpuSub() {
   493→    const sq = this.squads[1]
   494→    if (sq.subsUsed >= MATCH_RULES.maxSubs) return
   495→    if (this.minute() < 55) return
   496→    let worst: PlayerEntity | null = null
   497→    let worstIdx = -1
   498→    for (let i = 0; i < sq.players.length; i++) {
   499→      const p = sq.players[i]
   500→      if (p.isGK) continue
   501→      if (!worst || p.stamina < worst.stamina) {
   502→        worst = p
   503→        worstIdx = i
   504→      }
   505→    }
   506→    if (!worst || worst.stamina > 55) return
   507→    let bi = sq.bench.findIndex((b, i) => !sq.benchUsed[i] && b.role === worst!.data.role)
   508→    if (bi < 0) bi = sq.bench.findIndex((b, i) => !sq.benchUsed[i] && b.role !== 'GK')
   509→    if (bi >= 0) this.makeSubstitution(1, worstIdx, bi)
   510→  }

   514→  update(dt: number) {
   515→    this.excitement = Math.max(0, this.excitement - dt * 0.35)
   516→    this.audio.setExcitement(this.excitement * 0.8)
   517→    this.shotWindow = Math.max(0, this.shotWindow - dt)
   518→    if (this.shotWindow === 0) this.pendingShot = false
   519→
   520→    // cambios automáticos de la CPU cuando el cansancio aprieta
   521→    // (se revisan en toda fase: las bolas paradas son momento de cambios)
   522→    if (this.phase !== 'full') {
   523→      this.cpuSubTimer -= dt
   524→      if (this.cpuSubTimer <= 0) {
   525→        this.cpuSubTimer = 8
   526→        this.tryCpuSub()
   527→      }
   528→    }
   529→
   530→    switch (this.phase) {
   531→      case 'kickoff':
   532→      case 'restart':
   533→      case 'half': {
   534→        this.phaseTimer -= dt
   535→        if (this.phase === 'half') {
   536→          if (this.phaseTimer <= 0) {
   537→            this.half = 2
   538→            for (const sq of this.squads) sq.attackDir = (sq.attackDir * -1) as 1 | -1
   539→            // descanso: los jugadores recuperan resistencia en el mediempo
   540→            for (const sq of this.squads)
   541→              for (const p of sq.players) p.stamina = Math.min(100, p.stamina + MATCH_RULES.halfTimeRecovery)
   542→            this.setupKickoff(1)
   543→            this.addEvent('Comienza el segundo tiempo', 'info')
   544→          }
   545→          break
   546→        }
   547→        // congelados: solo animación
   548→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   549→        this.ballPrevX = this.ball.x
   550→        if (this.phaseTimer <= 0) {
   551→          this.phase = 'play'
   552→        }
   553→        break
   554→      }
   555→      case 'goal': {
   556→        this.phaseTimer -= dt
   557→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   558→        if (this.phaseTimer <= 0) {
   559→          this.celebrateScorer = null
   560→          this.celebrateTeam = null
   561→          this.setupKickoff(this.lastTouchTeamConceding())
   562→        }
   563→        break
   564→      }
   565→      case 'play':
   566→        this.updatePlay(dt)
   567→        break
   568→      case 'full':
   569→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   570→        break
   571→    }
   572→    this.input.endFrame()
   573→  }
   574→
   575→  private lastTouchTeamConceding(): TeamIndex {
   576→    if (!this.lastTouchPlayer) return this.kickoffTeam === 0 ? 1 : 0
   577→    // el equipo que recibió el gol saca
   578→    return this.lastTouchPlayer.team === 0 ? 1 : 0
   579→  }

   581→  private updatePlay(dt: number) {
   582→    this.clock += dt
   583→    this.ballPrevX = this.ball.x
   584→
   585→    this.updateControlledInput(dt)
   586→    updateMatchAI(this, dt)
   587→    for (const sq of this.squads) {
   588→      for (const p of sq.players) p.update(dt)
   589→    }
   590→    this.separatePlayers()
   591→    this.updateBallPossession(dt)
   592→    this.checkOutOfBounds()
   593→
   594→    if (this.owner) this.squads[this.owner.team].possession += dt
   595→
   596→    if (this.half === 1 && this.clock >= this.durationSec / 2) {
   597→      this.finishHalf()
   598→    } else if (this.half === 2 && this.clock >= this.durationSec && !this.finished) {
   599→      this.goFullTime()
   600→    }
   601→  }

   603→  /** Convierte la entrada humana en acciones del jugador controlado. */
   604→  private updateControlledInput(dt: number) {
   605→    const inp = this.input
   606→    // auto-cambio de jugador
   607→    const team0 = this.squads[0]
   608→    if (!this.owner || this.owner.team !== 0) {
   609→      this.switchHysteresis += dt
   610→      if (this.switchHysteresis > 0.18 || !this.controlled || this.controlled.team !== 0) {
   611→        const fut = this.ball.future(0.25)
   612→        let best: PlayerEntity | null = null
   613→        let bestD = Infinity
   614→        const nearOwnBox = Math.abs(this.ball.x - this.ownGoalOf(0)) < 20
   615→        for (const p of team0.players) {
   616→          if (p.isGK && !nearOwnBox) continue
   617→          const d = dist2d(p.x, p.z, fut.x, fut.z)
   618→          if (d < bestD) {
   619→            bestD = d
   620→            best = p
   621→          }
   622→        }
   623→        if (best && best !== this.controlled) this.controlled = best
   624→        this.switchHysteresis = 0
   625→      }
   626→    } else {
   627→      this.switchHysteresis = 0
   628→      this.controlled = this.owner
   629→    }
   630→
   631→    const c = this.controlled
   632→    if (!c) return
   633→    c.speedMult = this.diff.userBoost
   634→
   635→    // movimiento relativo a cámara (el engine lo traduce antes: aquí ya es mundo)
   636→    const mx = inp.move.x
   637→    const my = inp.move.y
   638→    const world = this.engineMoveToWorld(mx, my)
   639→    const moving = Math.hypot(world.x, world.z) > 0.1
   640→    c.sprinting = inp.sprint && moving
   641→    if (moving) {
   642→      const len = Math.hypot(world.x, world.z)
   643→      c.desiredVx = (world.x / len) * c.maxSpeed()
   644→      c.desiredVz = (world.z / len) * c.maxSpeed()
   645→    } else {
   646→      c.desiredVx = 0
   647→      c.desiredVz = 0
   648→    }
   649→
   650→    const isOwner = this.owner === c
   651→    if (isOwner) {
   652→      // disparo con carga
   653→      if (inp.isHeld('shoot')) {
   654→        this.charging = true
   655→        this.charge = Math.min(1, this.charge + dt / 0.85)
   656→        if (this.charge >= 1) {
   657→          this.releaseShot(c)
   658→        }
   659→      } else if (this.charging) {
   660→        this.releaseShot(c)
   661→      }
   662→      if (inp.consumePressed('pass')) {
   663→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   664→        this.doPass(c, aim.x, aim.z, false)
   665→        this.charge = 0
   666→        this.charging = false
   667→      }
   668→      if (inp.consumePressed('lob')) {
   669→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   670→        this.doPass(c, aim.x, aim.z, true)
   671→      }
   672→    } else {
   673→      // defensa
   674→      this.charge = 0
   675→      this.charging = false
   676→      if (inp.consumePressed('pass')) this.tryTackle(c)
   677→      if (inp.consumePressed('shoot')) {
   678→        const dx = moving ? world.x : Math.sin(c.dir)
   679→        const dz = moving ? world.z : Math.cos(c.dir)
   680→        this.slideTackle(c, dx, dz)
   681→      }
   682→      if (inp.consumePressed('lob') || inp.consumePressed('switch')) {
   683→        // cambio manual: siguiente más cercano
   684→        const fut = this.ball.future(0.25)
   685→        const others = team0.players
   686→          .filter((p) => !p.isGK && p !== c)
   687→          .sort((a, b) => dist2d(a.x, a.z, fut.x, fut.z) - dist2d(b.x, b.z, fut.x, fut.z))
   688→        if (others[0]) this.controlled = others[0]
   689→      }
   690→    }
   691→  }

   693→  private releaseShot(c: PlayerEntity) {
   694→    const world = this.engineMoveToWorld(this.input.move.x, this.input.move.y)
   695→    // input vertical en mundo: arriba = palo lejano, abajo = palo cercano
   696→    const aimProj = clamp(world.z, -1, 1)
   697→    this.doShoot(c, Math.max(0.25, this.charge), aimProj)
   698→    this.charge = 0
   699→    this.charging = false
   700→  }

   702→  /** Traduce input de pantalla a mundo según el modo de cámara (seteado por el engine). */
   703→  cameraMode: CameraMode = 'broadcast'
   704→  attackDirUser: 1 | -1 = 1
   705→  private engineMoveToWorld(x: number, y: number): { x: number; z: number } {
   706→    if (this.cameraMode === 'arco') {
   707→      // cámara tras el arco que ataca el usuario
   708→      const f = this.attackDirUser
   709→      return { x: y * f, z: -x * f }
   710→    }
   711→    return { x, z: -y }
   712→  }

   716→  private separatePlayers() {
   717→    const all = [...this.squads[0].players, ...this.squads[1].players]
   718→    for (let i = 0; i < all.length; i++) {
   719→      for (let j = i + 1; j < all.length; j++) {
   720→        const a = all[i]
   721→        const b = all[j]
   722→        const dx = b.x - a.x
   723→        const dz = b.z - a.z
   724→        const d = Math.hypot(dx, dz)
   725→        const min = PLAYER.radius * 2
   726→        if (d > 0.001 && d < min) {
   727→          const push = (min - d) / 2
   728→          const ux = dx / d
   729→          const uz = dz / d
   730→          a.x -= ux * push
   731→          a.z -= uz * push
   732→          b.x += ux * push
   733→          b.z += uz * push
   734→        }
   735→      }
   736→    }
   737→  }

   739→  private updateBallPossession(dt: number) {
   740→    const ball = this.ball
   741→    const postHit = ball.update(dt)
   742→    if (postHit === 'post') {
   743→      this.audio.post()
   744→      this.addEvent('¡Al palo!', 'chance')
   745→      this.excitement = Math.max(this.excitement, 0.6)
   746→    } else if (postHit === 'crossbar') {
   747→      this.audio.post()
   748→      this.addEvent('¡Al travesaño!', 'chance')
   749→      this.excitement = Math.max(this.excitement, 0.6)
   750→    }
   751→
   752→    const owner = this.owner
   753→    if (owner) {
   754→      if (owner.isGK && this.gkHold > 0) {
   755→        this.gkHold -= dt
   756→        // balón en las manos
   757→        ball.x = owner.x + Math.sin(owner.dir) * 0.35
   758→        ball.z = owner.z + Math.cos(owner.dir) * 0.35
   759→        ball.y = 1.05
   760→        ball.stop()
   761→        const gkTeam = owner.team
   762→        if (this.gkHold <= 0) {
   763→          // distribución automática del portero
   764→          const mate = this.pickGkDistribution(owner)
   765→          if (mate) {
   766→            const dist = dist2d(owner.x, owner.z, mate.x, mate.z)
   767→            const lofted = dist > 26
   768→            this.doPass(owner, mate.x - owner.x, mate.z - owner.z, lofted)
   769→            if (owner.team === 0) this.controlled = mate
   770→          } else {
   771→            this.kickBall(owner, this.squads[gkTeam].attackDir, 0, 16, 6)
   772→          }
   773→        }
   774→        return
   775→      }
   776→
   777→      // conducción: el balón "pega" al pie
   778→      this.ownerHold += dt
   779→      const ahead = owner.sprinting ? 0.95 : 0.55
   780→      const tx = owner.x + Math.sin(owner.dir) * ahead
   781→      const tz = owner.z + Math.cos(owner.dir) * ahead
   782→      ball.vx = (tx - ball.x) * 11
   783→      ball.vz = (tz - ball.z) * 11
   784→      ball.vy = 0
   785→      ball.y = BALL.r
   786→      if (dist2d(ball.x, ball.z, owner.x, owner.z) > 2.4) this.loseControl()
   787→      return
   788→    }
   789→
   790→    // balón libre: intento de control por proximidad
   791→    if (ball.y < PLAYER.controlHeight) {
   792→      const all = [...this.squads[0].players, ...this.squads[1].players]
   793→      let best: PlayerEntity | null = null
   794→      let bestD = Infinity
   795→      for (const p of all) {
   796→        if (p.kickCooldown > 0 || p.stunTimer > 0 || p.slideTimer > 0) continue
   797→        const d = dist2d(p.x, p.z, ball.x, ball.z)
   798→        if (d < PLAYER.reach && d < bestD) {
   799→          bestD = d
   800→          best = p
   801→        }
   802→      }
   803→      if (best) {
   804→        const sp = ball.speed
   805→        const controllable = sp < 9 || Math.random() < 0.35 + best.data.skill * 0.55
   806→        if (sp > 16 && best.isGK && this.isBallThreat(best)) {
   807→          // atajada
   808→          this.gainControl(best)
   809→          this.squads[best.team].onTarget += this.pendingShot && this.shotTeam !== best.team ? 1 : 0
   810→          if (this.pendingShot && this.shotTeam !== best.team) {
   811→            this.addEvent(`¡Atajada de ${best.data.name}!`, 'chance')
   812→            this.excitement = Math.max(this.excitement, 0.7)
   813→            this.audio.ooh()
   814→          }
   815→          this.pendingShot = false
   816→          this.audio.catchBall()
   817→          return
   818→        }
   819→        if (controllable) {
   820→          this.gainControl(best)
   821→          ball.vx *= 0.15
   822→          ball.vz *= 0.15
   823→          ball.vy = 0
   824→        } else {
   825→          // control malo: rebota
   826→          ball.vx = ball.vx * -0.25 + (Math.random() - 0.5) * 4
   827→          ball.vz = ball.vz * -0.25 + (Math.random() - 0.5) * 4
   828→          ball.lastTouch = best.team
   829→          this.lastTouchPlayer = best
   830→          best.kickCooldown = 0.25
   831→        }
   832→      }
   833→    }
   834→  }

   836→  private isBallThreat(gk: PlayerEntity): boolean {
   837→    // el balón va hacia su arco rápido
   838→    const gx = this.ownGoalOf(gk.team)
   839→    const toward = gx < PITCH.L / 2 ? this.ball.vx < -6 : this.ball.vx > 6
   840→    return toward && this.ball.speed > 12
   841→  }

   843→  private pickGkDistribution(gk: PlayerEntity): PlayerEntity | null {
   844→    const squad = this.squads[gk.team]
   845→    let best: PlayerEntity | null = null
   846→    let bestScore = -Infinity
   847→    for (const mate of squad.players) {
   848→      if (mate === gk) continue
   849→      const d = dist2d(gk.x, gk.z, mate.x, mate.z)
   850→      const open = Math.min(9, this.closestOppDist(mate))
   851→      const score = open * 2 - d * 0.06 + (mate.data.role === 'DF' ? 2 : 0)
   852→      if (score > bestScore) {
   853→        bestScore = score
   854→        best = mate
   855→      }
   856→    }
   857→    return best
   858→  }

   862→  private checkOutOfBounds() {
   863→    const b = this.ball
   864→    const prevX = this.ballPrevX
   865→
   866→    // gol: cruza la línea de meta por debajo del travesaño y dentro de la boca
   867→    if (prevX > 0 && b.x <= 0 && Math.abs(b.z - GOAL.zCenter) < PITCH.goalW / 2 && b.y < PITCH.goalH) {
   868→      this.handleGoalByGoalLine(0)
   869→      return
   870→    }
   871→    if (prevX < PITCH.L && b.x >= PITCH.L && Math.abs(b.z - GOAL.zCenter) < PITCH.goalW / 2 && b.y < PITCH.goalH) {
   872→      this.handleGoalByGoalLine(PITCH.L)
   873→      return
   874→    }
   875→
   876→    // líneas de banda
   877→    if (b.z <= 0 || b.z >= PITCH.W) {
   878→      const team = b.lastTouch === 0 ? 1 : 0
   879→      this.lastOutWasNearMiss = false
   880→      if (this.pendingShot) {
   881→        this.audio.ooh()
   882→        this.excitement = Math.max(this.excitement, 0.5)
   883→        this.lastOutWasNearMiss = true
   884→      }
   885→      this.setRestart({ type: 'throwin', team, x: clamp(b.x, 1.5, PITCH.L - 1.5), z: b.z <= 0 ? 0.3 : PITCH.W - 0.3 })
   886→      this.addEvent(this.lastOutWasNearMiss ? '¡Uy! Salió muy cerca del arco' : 'Saque de banda', 'info')
   887→      this.pendingShot = false
   888→      return
   889→    }
   890→
   891→    // líneas de meta (sin gol)
   892→    if (b.x <= 0 || b.x >= PITCH.L) {
   893→      const leftSide = b.x <= 0
   894→      const defendingTeam: TeamIndex = this.ownGoalOf(0) === (leftSide ? 0 : PITCH.L) ? 0 : 1
   895→      const attackTeam: TeamIndex = defendingTeam === 0 ? 1 : 0
   896→      if (b.lastTouch === defendingTeam) {
   897→        // córner
   898→        const z = b.z <= PITCH.W / 2 ? 0.4 : PITCH.W - 0.4
   899→        this.setRestart({ type: 'corner', team: attackTeam, x: leftSide ? 0.4 : PITCH.L - 0.4, z })
   900→        this.addEvent(`Córner para ${this.squads[attackTeam].roster.def.short}`, 'info')
   901→      } else {
   902→        // saque de puerta
   903→        const z = clamp(GOAL.zCenter + (b.z - GOAL.zCenter) * 0.3, GOAL.zMin + 2, GOAL.zMax - 2)
   904→        this.setRestart({
   905→          type: 'goalkick',
   906→          team: defendingTeam,
   907→          x: leftSide ? PITCH.sixD : PITCH.L - PITCH.sixD,
   908→          z,
   909→        })
   910→        this.addEvent('Saque de puerta', 'info')
   911→      }
   912→      this.pendingShot = false
   913→    }
   914→  }
   915→
   916→  private handleGoalByGoalLine(side: 0 | 105) {
   917→    // el arco en x=side lo defiende el equipo cuyo arco propio es ese lado
   918→    const defender: TeamIndex = this.ownGoalOf(0) === side ? 0 : 1
   919→    const scorer: TeamIndex = defender === 0 ? 1 : 0
   920→    this.handleGoal(scorer)
   921→  }
