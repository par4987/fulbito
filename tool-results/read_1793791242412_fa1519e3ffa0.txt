   179→   179→  private placeFormation(sq: Squad, kickingTeam: TeamIndex) {
   180→   180→    for (const p of sq.players) {
   181→   181→      let wx = sq.attackDir > 0 ? p.ax * PITCH.L : (1 - p.ax) * PITCH.L
   182→   182→      let wz = p.az * PITCH.W
   183→   183→      // todos en su propia mitad en el saque
   184→   184→      if (sq.attackDir > 0) wx = Math.min(wx, PITCH.L / 2 - 0.5)
   185→   185→      else wx = Math.max(wx, PITCH.L / 2 + 0.5)
   186→   186→      // rivales fuera del círculo central
   187→   187→      if (sq.attackDir !== this.squads[kickingTeam].attackDir) {
   188→   188→        const d = dist2d(wx, wz, PITCH.L / 2, PITCH.W / 2)
   189→   189→        if (d < PITCH.circleR + 1) {
   190→   190→          const ux = (wx - PITCH.L / 2) / (d || 1)
   191→   191→          const uz = (wz - PITCH.W / 2) / (d || 1)
   192→   192→          wx = PITCH.L / 2 + ux * (PITCH.circleR + 1.5)
   193→   193→          wz = PITCH.W / 2 + uz * (PITCH.circleR + 1.5)
   194→   194→        }
   195→   195→      }
   196→   196→      p.teleport(clamp(wx, 1, PITCH.L - 1), clamp(wz, 1, PITCH.W - 1))
   197→   197→      p.targetX = p.x
   198→   198→      p.targetZ = p.z
   199→   199→    }
   200→   200→  }

   213→   213→      const d = dist2d(p.x, p.z, info.x, info.z)
   214→   214→      if (d < bestD) {
   215→   215→        bestD = d
   216→   216→        best = p
   217→   217→      }
   218→   218→    }
   219→   219→    if (best) {
   220→   220→      best.teleport(info.x - Math.sign(info.x - PITCH.L / 2) * 0.9, info.z)
   221→   221→      this.gainControl(best)
   222→   222→      if (info.team === 0) this.controlled = best
   223→   223→      else this.controlled = null
   224→   224→    }

   312→   312→  /** Pase raso o alto hacia el mejor compañero según dirección deseada. */
   313→   313→  doPass(p: PlayerEntity, aimX: number, aimZ: number, lofted = false) {
   314→   314→    if (p.kickCooldown > 0) return
   315→   315→    const receiver = this.pickPassTarget(p, aimX, aimZ, lofted)
   316→   316→    if (!receiver) {
   317→   317→      // despeje hacia adelante
   318→   318→      const dir = this.squads[p.team].attackDir
   319→   319→      this.kickBall(p, dir, (Math.random() - 0.5) * 20, 16, 4)
   320→   320→      return
   321→   321→    }
   322→   322→    const err = (1 - p.data.skill) * this.diff.passErr * (lofted ? 1.6 : 1)
   323→   323→    const noise = (Math.random() - 0.5) * 2 * err
   324→   324→    const cs = Math.cos(noise)
   325→   325→    const sn = Math.sin(noise)
   326→   326→    const dist = dist2d(p.x, p.z, receiver.x, receiver.z)
   327→   327→    const speed = lofted ? clamp(10 + dist * 0.42, 13, 20) : clamp(9 + dist * 0.55, 10, 21)
   328→   328→    // punto adelantado hacia donde corre el receptor
   329→   329→    const lead = lofted ? 0.35 : 0.45
   330→   330→    let dx = receiver.x + receiver.vx * lead - p.x
   331→   331→    let dz = receiver.z + receiver.vz * lead - p.z
   332→   332→    const rx = dx * cs - dz * sn
   333→   333→    const rz = dx * sn + dz * cs
   334→   334→    const vz = lofted ? clamp(dist * 0.32, 3, 8.5) : 0.35
   335→   335→    this.kickBall(p, rx, rz, speed, vz)
   336→   336→    if (p.team === 0 && !lofted) this.controlled = receiver
   337→   337→    if (p.team === 1) this.aiPassHint = receiver
   338→   338→  }

   340→   340→  private pickPassTarget(p: PlayerEntity, aimX: number, aimZ: number, lofted: boolean): PlayerEntity | null {
   341→   341→    const aimLen = Math.hypot(aimX, aimZ)
   342→   342→    const nx = aimLen > 0.1 ? aimX / aimLen : Math.sin(p.dir)
   343→   343→    const nz = aimLen > 0.1 ? aimZ / aimLen : Math.cos(p.dir)
   344→   344→    const squad = this.squads[p.team]
   345→   345→    let best: PlayerEntity | null = null
   346→   346→    let bestScore = -Infinity
   347→   347→    for (const mate of squad.players) {
   348→   348→      if (mate === p || mate.isGK) continue
   349→   349→      const dx = mate.x - p.x
   350→   350→      const dz = mate.z - p.z
   351→   351→      const d = Math.hypot(dx, dz)
   352→   352→      if (d < 2.5 || d > (lofted ? 55 : 34)) continue
   353→   353→      const dot = (dx / d) * nx + (dz / d) * nz
   354→   354→      if (!lofted && dot < 0.2) continue
   355→   355→      const open = this.closestOppDist(mate)
   356→   356→      let lanePen = 0
   357→   357→      if (!lofted) lanePen = this.laneBlocked(p, mate) ? 2.5 : 0
   358→   358→      const score = dot * 3 + open * 0.35 - d * 0.045 - lanePen
   359→   359→      if (score > bestScore) {
   360→   360→        bestScore = score
   361→   361→        best = mate
   362→   362→      }
   363→   363→    }
   364→   364→    return best
   365→   365→  }

   367→   367→  private closestOppDist(p: PlayerEntity): number {
   368→   368→    const opp = this.squads[p.team === 0 ? 1 : 0].players
   369→   369→    let d = Infinity
   370→   370→    for (const o of opp) d = Math.min(d, dist2d(o.x, o.z, p.x, p.z))
   371→   371→    return d
   372→   372→  }
   373→   373→
   374→   374→  private laneBlocked(from: PlayerEntity, to: PlayerEntity): boolean {
   375→   375→    const opp = this.squads[from.team === 0 ? 1 : 0].players
   376→   376→    const dx = to.x - from.x
   377→   377→    const dz = to.z - from.z
   378→   378→    const len = Math.hypot(dx, dz) || 1
   379→   379→    for (const o of opp) {
   380→   380→      const t = clamp(((o.x - from.x) * dx + (o.z - from.z) * dz) / (len * len), 0.1, 0.9)
   381→   381→      const px = from.x + dx * t
   382→   382→      const pz = from.z + dz * t
   383→   383→      if (dist2d(o.x, o.z, px, pz) < 1.3) return true
   384→   384→    }
   385→   385→    return false
   386→   386→  }

   388→   388→  /** Disparo al arco. power 0..1, aim = componente perpendicular del input. */
   389→   389→  doShoot(p: PlayerEntity, power: number, aim: number) {
   390→   390→    if (p.kickCooldown > 0) return
   391→   391→    const gx = this.goalOf(p.team)
   392→   392→    const dirX = gx > p.x ? 1 : -1
   393→   393→    const dist = Math.abs(gx - p.x)
   394→   394→    const noise = (1 - p.data.skill) * this.diff.shootErr * (2 + dist * 0.06) + power * 0.04
   395→   395→    const angle = (Math.random() - 0.5) * 2 * noise
   396→   396→    let dz = GOAL.zCenter - p.z + aim * 3.4
   397→   397→    let dx = dirX * Math.max(8, dist)
   398→   398→    const cs = Math.cos(angle)
   399→   399→    const sn = Math.sin(angle)
   400→   400→    const rx = dx * cs - dz * sn
   401→   401→    const rz = dx * sn + dz * cs
   402→   402→    const speed = 18 + power * 12
   403→   403→    const vz = 0.5 + power * 3.1
   404→   404→    this.kickBall(p, rx, rz, speed, vz)
   405→   405→    this.squads[p.team].shots++
   406→   406→    this.pendingShot = true
   407→   407→    this.shotTeam = p.team
   408→   408→    this.shotWindow = 1.3
   409→   409→  }

   411→   411→  /** Entrada (presión): intenta robar el balón al dueño cercano. */
   412→   412→  tryTackle(p: PlayerEntity) {
   413→   413→    if (p.tackleCd > 0 || p.stunTimer > 0) return
   414→   414→    p.tackleCd = PLAYER.tackleCooldown
   415→   415→    const owner = this.owner
   416→   416→    if (owner && owner.team !== p.team && dist2d(p.x, p.z, owner.x, owner.z) < PLAYER.tackleRange) {
   417→   417→      const prob = 0.45 + p.data.skill * 0.25 + this.diff.tackle * 0.15 - owner.data.skill * 0.2
   418→   418→      if (Math.random() < prob) {
   419→   419→        this.loseControl()
   420→   420→        this.ball.kick(p.vx + Math.sin(p.dir) * 4, p.vz + Math.cos(p.dir) * 4, 7, 0.4)
   421→   421→        this.ball.lastTouch = p.team
   422→   422→        this.lastTouchPlayer = p
   423→   423→        owner.stunTimer = 0.3
   424→   424→        this.audio.kick(0.3)
   425→   425→      } else {
   426→   426→        p.stunTimer = 0.35
   427→   427→      }
   428→   428→    } else if (!owner && dist2d(p.x, p.z, this.ball.x, this.ball.z) < PLAYER.tackleRange) {
   429→   429→      // empujón al balón libre
   430→   430→      this.ball.kick(Math.sin(p.dir), Math.cos(p.dir), 8, 0.5)
   431→   431→      this.ball.lastTouch = p.team
   432→   432→      this.lastTouchPlayer = p
   433→   433→      p.kickCooldown = 0.3
   434→   434→    }
   435→   435→  }

   437→   437→  slideTackle(p: PlayerEntity, dx: number, dz: number) {

   472→   472→  makeSubstitution(team: TeamIndex, outIdx: number, benchIdx: number): SubResult {
   473→   473→    if (this.phase === 'full') return 'invalido'
   474→   474→    const sq = this.squads[team]

   476→   476→    const out = sq.players[outIdx]
   477→   477→    const sub = sq.bench[benchIdx]
   478→   478→    if (!out || !sub || sq.benchUsed[benchIdx]) return 'invalido'
   479→   479→    if (out.isGK !== (sub.role === 'GK')) return 'invalido'
   480→   480→    const oldName = out.data.name
   481→   481→    out.data = sub
   482→   482→    out.stamina = 100
   483→   483→    sq.benchUsed[benchIdx] = true
   484→   484→    sq.subsUsed++
   485→   485→    this.addEvent(`Cambio ${sq.roster.def.short}: entra ${sub.name} por ${oldName}`, 'info')
   486→   486→    return 'ok'

   491→   491→  /** La CPU reemplaza a su jugador de campo más cansado a partir del minuto 55. */
   492→   492→  private tryCpuSub() {
   493→   493→    const sq = this.squads[1]
   494→   494→    if (sq.subsUsed >= MATCH_RULES.maxSubs) return
   495→   495→    if (this.minute() < 55) return
   496→   496→    let worst: PlayerEntity | null = null
   497→   497→    let worstIdx = -1
   498→   498→    for (let i = 0; i < sq.players.length; i++) {
   499→   499→      const p = sq.players[i]
   500→   500→      if (p.isGK) continue
   501→   501→      if (!worst || p.stamina < worst.stamina) {
   502→   502→        worst = p
   503→   503→        worstIdx = i
   504→   504→      }
   505→   505→    }
   506→   506→    if (!worst || worst.stamina > 55) return
   507→   507→    let bi = sq.bench.findIndex((b, i) => !sq.benchUsed[i] && b.role === worst!.data.role)
   508→   508→    if (bi < 0) bi = sq.bench.findIndex((b, i) => !sq.benchUsed[i] && b.role !== 'GK')
   509→   509→    if (bi >= 0) this.makeSubstitution(1, worstIdx, bi)
   510→   510→  }

   514→   514→  update(dt: number) {
   515→   515→    this.excitement = Math.max(0, this.excitement - dt * 0.35)
   516→   516→    this.audio.setExcitement(this.excitement * 0.8)
   517→   517→    this.shotWindow = Math.max(0, this.shotWindow - dt)
   518→   518→    if (this.shotWindow === 0) this.pendingShot = false
   519→   519→
   520→   520→    // cambios automáticos de la CPU cuando el cansancio aprieta
   521→   521→    // (se revisan en toda fase: las bolas paradas son momento de cambios)
   522→   522→    if (this.phase !== 'full') {
   523→   523→      this.cpuSubTimer -= dt
   524→   524→      if (this.cpuSubTimer <= 0) {
   525→   525→        this.cpuSubTimer = 8
   526→   526→        this.tryCpuSub()
   527→   527→      }
   528→   528→    }
   529→   529→
   530→   530→    switch (this.phase) {
   531→   531→      case 'kickoff':
   532→   532→      case 'restart':
   533→   533→      case 'half': {
   534→   534→        this.phaseTimer -= dt
   535→   535→        if (this.phase === 'half') {
   536→   536→          if (this.phaseTimer <= 0) {
   537→   537→            this.half = 2
   538→   538→            for (const sq of this.squads) sq.attackDir = (sq.attackDir * -1) as 1 | -1
   539→   539→            // descanso: los jugadores recuperan resistencia en el mediempo
   540→   540→            for (const sq of this.squads)
   541→   541→              for (const p of sq.players) p.stamina = Math.min(100, p.stamina + MATCH_RULES.halfTimeRecovery)
   542→   542→            this.setupKickoff(1)
   543→   543→            this.addEvent('Comienza el segundo tiempo', 'info')
   544→   544→          }
   545→   545→          break
   546→   546→        }
   547→   547→        // congelados: solo animación
   548→   548→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   549→   549→        this.ballPrevX = this.ball.x
   550→   550→        if (this.phaseTimer <= 0) {
   551→   551→          this.phase = 'play'
   552→   552→        }
   553→   553→        break
   554→   554→      }
   555→   555→      case 'goal': {
   556→   556→        this.phaseTimer -= dt
   557→   557→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   558→   558→        if (this.phaseTimer <= 0) {
   559→   559→          this.celebrateScorer = null
   560→   560→          this.celebrateTeam = null
   561→   561→          this.setupKickoff(this.lastTouchTeamConceding())
   562→   562→        }
   563→   563→        break
   564→   564→      }
   565→   565→      case 'play':
   566→   566→        this.updatePlay(dt)
   567→   567→        break
   568→   568→      case 'full':
   569→   569→        for (const sq of this.squads) for (const p of sq.players) { p.desiredVx = 0; p.desiredVz = 0; p.update(dt) }
   570→   570→        break
   571→   571→    }
   572→   572→    this.input.endFrame()
   573→   573→  }

   581→   581→  private updatePlay(dt: number) {

   583→   583→    this.ballPrevX = this.ball.x
   584→   584→
   585→   585→    this.updateControlledInput(dt)
   586→   586→    updateMatchAI(this, dt)
   587→   587→    for (const sq of this.squads) {
   588→   588→      for (const p of sq.players) p.update(dt)
   589→   589→    }
   590→   590→    this.separatePlayers()
   591→   591→    this.updateBallPossession(dt)
   592→   592→    this.checkOutOfBounds()
   593→   593→
   594→   594→    if (this.owner) this.squads[this.owner.team].possession += dt
   595→   595→
   596→   596→    if (this.half === 1 && this.clock >= this.durationSec / 2) {
   597→   597→      this.finishHalf()
   598→   598→    } else if (this.half === 2 && this.clock >= this.durationSec && !this.finished) {
   599→   599→      this.goFullTime()
   600→   600→    }

   603→   603→  /** Convierte la entrada humana en acciones del jugador controlado. */
   604→   604→  private updateControlledInput(dt: number) {
   605→   605→    const inp = this.input
   606→   606→    // auto-cambio de jugador
   607→   607→    const team0 = this.squads[0]
   608→   608→    if (!this.owner || this.owner.team !== 0) {
   609→   609→      this.switchHysteresis += dt
   610→   610→      if (this.switchHysteresis > 0.18 || !this.controlled || this.controlled.team !== 0) {
   611→   611→        const fut = this.ball.future(0.25)
   612→   612→        let best: PlayerEntity | null = null
   613→   613→        let bestD = Infinity
   614→   614→        const nearOwnBox = Math.abs(this.ball.x - this.ownGoalOf(0)) < 20
   615→   615→        for (const p of team0.players) {
   616→   616→          if (p.isGK && !nearOwnBox) continue
   617→   617→          const d = dist2d(p.x, p.z, fut.x, fut.z)
   618→   618→          if (d < bestD) {
   619→   619→            bestD = d
   620→   620→            best = p
   621→   621→          }
   622→   622→        }
   623→   623→        if (best && best !== this.controlled) this.controlled = best
   624→   624→        this.switchHysteresis = 0
   625→   625→      }
   626→   626→    } else {
   627→   627→      this.switchHysteresis = 0
   628→   628→      this.controlled = this.owner
   629→   629→    }
   630→   630→
   631→   631→    const c = this.controlled
   632→   632→    if (!c) return
   633→   633→    c.speedMult = this.diff.userBoost
   634→   634→
   635→   635→    // movimiento relativo a cámara (el engine lo traduce antes: aquí ya es mundo)
   636→   636→    const mx = inp.move.x
   637→   637→    const my = inp.move.y
   638→   638→    const world = this.engineMoveToWorld(mx, my)
   639→   639→    const moving = Math.hypot(world.x, world.z) > 0.1
   640→   640→    c.sprinting = inp.sprint && moving
   641→   641→    if (moving) {
   642→   642→      const len = Math.hypot(world.x, world.z)
   643→   643→      c.desiredVx = (world.x / len) * c.maxSpeed()
   644→   644→      c.desiredVz = (world.z / len) * c.maxSpeed()
   645→   645→    } else {
   646→   646→      c.desiredVx = 0
   647→   647→      c.desiredVz = 0
   648→   648→    }
   649→   649→
   650→   650→    const isOwner = this.owner === c
   651→   651→    if (isOwner) {
   652→   652→      // disparo con carga
   653→   653→      if (inp.isHeld('shoot')) {
   654→   654→        this.charging = true
   655→   655→        this.charge = Math.min(1, this.charge + dt / 0.85)
   656→   656→        if (this.charge >= 1) {
   657→   657→          this.releaseShot(c)
   658→   658→        }
   659→   659→      } else if (this.charging) {
   660→   660→        this.releaseShot(c)
   661→   661→      }
   662→   662→      if (inp.consumePressed('pass')) {
   663→   663→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   664→   664→        this.doPass(c, aim.x, aim.z, false)
   665→   665→        this.charge = 0
   666→   666→        this.charging = false
   667→   667→      }
   668→   668→      if (inp.consumePressed('lob')) {
   669→   669→        const aim = moving ? world : { x: Math.sin(c.dir), z: Math.cos(c.dir) }
   670→   670→        this.doPass(c, aim.x, aim.z, true)
   671→   671→      }
   672→   672→    } else {
   673→   673→      // defensa
   674→   674→      this.charge = 0
   675→   675→      this.charging = false
   676→   676→      if (inp.consumePressed('pass')) this.tryTackle(c)
   677→   677→      if (inp.consumePressed('shoot')) {
   678→   678→        const dx = moving ? world.x : Math.sin(c.dir)
   679→   679→        const dz = moving ? world.z : Math.cos(c.dir)
   680→   680→        this.slideTackle(c, dx, dz)
   681→   681→      }
   682→   682→      if (inp.consumePressed('lob') || inp.consumePressed('switch')) {
   683→   683→        // cambio manual: siguiente más cercano
   684→   684→        const fut = this.ball.future(0.25)
   685→   685→        const others = team0.players
   686→   686→          .filter((p) => !p.isGK && p !== c)
   687→   687→          .sort((a, b) => dist2d(a.x, a.z, fut.x, fut.z) - dist2d(b.x, b.z, fut.x, fut.z))
   688→   688→        if (others[0]) this.controlled = others[0]
   689→   689→      }
   690→   690→    }
   691→   691→  }

   698→   698→    this.charge = 0
   699→   699→    this.charging = false
   700→   700→  }

   703→   703→  cameraMode: CameraMode = 'broadcast'

   705→   705→  private engineMoveToWorld(x: number, y: number): { x: number; z: number } {
   706→   706→    if (this.cameraMode === 'arco') {
   707→   707→      // cámara tras el arco que ataca el usuario
   708→   708→      const f = this.attackDirUser
   709→   709→      return { x: y * f, z: -x * f }
   710→   710→    }
   711→   711→    return { x, z: -y }

   716→   716→  private separatePlayers() {
   717→   717→    const all = [...this.squads[0].players, ...this.squads[1].players]
   718→   718→    for (let i = 0; i < all.length; i++) {
   719→   719→      for (let j = i + 1; j < all.length; j++) {
   720→   720→        const a = all[i]
   721→   721→        const b = all[j]
   722→   722→        const dx = b.x - a.x
   723→   723→        const dz = b.z - a.z
   724→   724→        const d = Math.hypot(dx, dz)
   725→   725→        const min = PLAYER.radius * 2
   726→   726→        if (d > 0.001 && d < min) {
   727→   727→          const push = (min - d) / 2
   728→   728→          const ux = dx / d
   729→   729→          const uz = dz / d
   730→   730→          a.x -= ux * push
   731→   731→          a.z -= uz * push
   732→   732→          b.x += ux * push
   733→   733→          b.z += uz * push
   734→   734→        }
   735→   735→      }
   736→   736→    }
   737→   737→  }

   739→   739→  private updateBallPossession(dt: number) {
   740→   740→    const ball = this.ball
   741→   741→    const postHit = ball.update(dt)
   742→   742→    if (postHit === 'post') {
   743→   743→      this.audio.post()
   744→   744→      this.addEvent('¡Al palo!', 'chance')
   745→   745→      this.excitement = Math.max(this.excitement, 0.6)
   746→   746→    } else if (postHit === 'crossbar') {
   747→   747→      this.audio.post()
   748→   748→      this.addEvent('¡Al travesaño!', 'chance')
   749→   749→      this.excitement = Math.max(this.excitement, 0.6)
   750→   750→    }
   751→   751→
   752→   752→    const owner = this.owner
   753→   753→    if (owner) {
   754→   754→      if (owner.isGK && this.gkHold > 0) {
   755→   755→        this.gkHold -= dt
   756→   756→        // balón en las manos
   757→   757→        ball.x = owner.x + Math.sin(owner.dir) * 0.35
   758→   758→        ball.z = owner.z + Math.cos(owner.dir) * 0.35
   759→   759→        ball.y = 1.05
   760→   760→        ball.stop()
   761→   761→        const gkTeam = owner.team
   762→   762→        if (this.gkHold <= 0) {
   763→   763→          // distribución automática del portero
   764→   764→          const mate = this.pickGkDistribution(owner)
   765→   765→          if (mate) {
   766→   766→            const dist = dist2d(owner.x, owner.z, mate.x, mate.z)
   767→   767→            const lofted = dist > 26
   768→   768→            this.doPass(owner, mate.x - owner.x, mate.z - owner.z, lofted)
   769→   769→            if (owner.team === 0) this.controlled = mate
   770→   770→          } else {
   771→   771→            this.kickBall(owner, this.squads[gkTeam].attackDir, 0, 16, 6)
   772→   772→          }
   773→   773→        }
   774→   774→        return
   775→   775→      }
   776→   776→
   777→   777→      // conducción: el balón "pega" al pie
   778→   778→      this.ownerHold += dt
   779→   779→      const ahead = owner.sprinting ? 0.95 : 0.55
   780→   780→      const tx = owner.x + Math.sin(owner.dir) * ahead
   781→   781→      const tz = owner.z + Math.cos(owner.dir) * ahead
   782→   782→      ball.vx = (tx - ball.x) * 11
   783→   783→      ball.vz = (tz - ball.z) * 11
   784→   784→      ball.vy = 0
   785→   785→      ball.y = BALL.r
   786→   786→      if (dist2d(ball.x, ball.z, owner.x, owner.z) > 2.4) this.loseControl()
   787→   787→      return
   788→   788→    }
   789→   789→
   790→   790→    // balón libre: intento de control por proximidad
   791→   791→    if (ball.y < PLAYER.controlHeight) {
   792→   792→      const all = [...this.squads[0].players, ...this.squads[1].players]
   793→   793→      let best: PlayerEntity | null = null
   794→   794→      let bestD = Infinity
   795→   795→      for (const p of all) {
   796→   796→        if (p.kickCooldown > 0 || p.stunTimer > 0 || p.slideTimer > 0) continue
   797→   797→        const d = dist2d(p.x, p.z, ball.x, ball.z)
   798→   798→        if (d < PLAYER.reach && d < bestD) {
   799→   799→          bestD = d
   800→   800→          best = p
   801→   801→        }
   802→   802→      }
   803→   803→      if (best) {
   804→   804→        const sp = ball.speed
   805→   805→        const controllable = sp < 9 || Math.random() < 0.35 + best.data.skill * 0.55
   806→   806→        if (sp > 16 && best.isGK && this.isBallThreat(best)) {
   807→   807→          // atajada
   808→   808→          this.gainControl(best)
   809→   809→          this.squads[best.team].onTarget += this.pendingShot && this.shotTeam !== best.team ? 1 : 0
   810→   810→          if (this.pendingShot && this.shotTeam !== best.team) {
   811→   811→            this.addEvent(`¡Atajada de ${best.data.name}!`, 'chance')
   812→   812→            this.excitement = Math.max(this.excitement, 0.7)
   813→   813→            this.audio.ooh()
   814→   814→          }
   815→   815→          this.pendingShot = false
   816→   816→          this.audio.catchBall()
   817→   817→          return
   818→   818→        }
   819→   819→        if (controllable) {
   820→   820→          this.gainControl(best)
   821→   821→          ball.vx *= 0.15
   822→   822→          ball.vz *= 0.15
   823→   823→          ball.vy = 0
   824→   824→        } else {
   825→   825→          // control malo: rebota
   826→   826→          ball.vx = ball.vx * -0.25 + (Math.random() - 0.5) * 4
   827→   827→          ball.vz = ball.vz * -0.25 + (Math.random() - 0.5) * 4
   828→   828→          ball.lastTouch = best.team
   829→   829→          this.lastTouchPlayer = best
   830→   830→          best.kickCooldown = 0.25
   831→   831→        }
   832→   832→      }
   833→   833→    }
   834→   834→  }

   836→   836→  private isBallThreat(gk: PlayerEntity): boolean {
   837→   837→    // el balón va hacia su arco rápido
   838→   838→    const gx = this.ownGoalOf(gk.team)
   839→   839→    const toward = gx < PITCH.L / 2 ? this.ball.vx < -6 : this.ball.vx > 6
   840→   840→    return toward && this.ball.speed > 12
   841→   841→  }

   843→   843→  private pickGkDistribution(gk: PlayerEntity): PlayerEntity | null {
   844→   844→    const squad = this.squads[gk.team]
   845→   845→    let best: PlayerEntity | null = null
   846→   846→    let bestScore = -Infinity
   847→   847→    for (const mate of squad.players) {
   848→   848→      if (mate === gk) continue
   849→   849→      const d = dist2d(gk.x, gk.z, mate.x, mate.z)
   850→   850→      const open = Math.min(9, this.closestOppDist(mate))
   851→   851→      const score = open * 2 - d * 0.06 + (mate.data.role === 'DF' ? 2 : 0)
   852→   852→      if (score > bestScore) {
   853→   853→        bestScore = score
   854→   854→        best = mate
   855→   855→      }
   856→   856→    }
   857→   857→    return best
   858→   858→  }

   862→   862→  private checkOutOfBounds() {
   863→   863→    const b = this.ball

   901→   901→      } else {

   906→   906→          team: defendingTeam,

   908→   908→          z,
   909→   909→        })

   911→   911→      }
   912→   912→      this.pendingShot = false
   913→   913→    }
